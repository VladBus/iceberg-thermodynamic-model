                          !DO I=1,IJS1
      APX2=0.             !APX2(I,1)=0.

      J1=J+1

      CI1=0.

      ELSE
      CI1=AN3(I1,J)
      I3=I+2
      ENDIF
      IF(IDY(I,J).EQ.0) THEN
      CI2=0.
      ELSE
      CI2=AN3(I-1,J)
      ENDIF
      IF(IX1.EQ.0) THEN
      CJ1=0.
      J3=J1
      ELSE
      CJ1=AN3(I,J1)
      J3=J+2
      ENDIF
      IF(IDX(I,J).EQ.0) THEN
      CJ2=0.
      ELSE
      CJ2=AN3(I,J-1)
      ENDIF
!----------------------------------------------------
      CDX=0.
      IF(UU.GT.0.) THEN
      CDX=(CC-CJ2)*UU
      ELSE
      CDX=(CJ1-CC)*UU
      ENDIF
      IF(CDX.NE.0.) GOTO 201
      IF(UM.GT.0.)  CDX=(CC-CJ2)*UM
      IF(UP.LT.0.)  CDX=(CJ1-CC)*UP
  201 CONTINUE
      CDX=CDX+CC*(UP-UM)*0.5
!----------------------------------------------------
      CDY=0.
      IF(VV.GT.0.) THEN
      CDY=(CC-CI1)*VV
      ELSE
      CDY=(CI2-CC)*VV
      ENDIF
      IF(CDY.NE.0.) GOTO 202
      IF(VP.GT.0.)  CDY=(CC-CI1)*VP
      IF(VM.LT.0.)  CDY=(CI2-CC)*VM
  202 CONTINUE
      CDY=CDY+CC*(VM-VP)*0.5
!------------------------------------------------------
      CD2(I,J)=CC-DT*(CDX+CDY)/DX
!--------------------------------------------------
      IF(IX1.NE.0) THEN
      IF(UP.LT.0.) THEN
      FLXP=CJ1*UP
      ELSE
      FLXP=CC*UP
      ENDIF
      IF(J.NE.JS) THEN
      J3=J+2

      CJ3=AN3(I,J3)
      ELSE
      CJ3=0.
      ENDIF
      ENDIF

      ENDIF
      IF(IY1.NE.0) THEN
         IF(VP.LT.0.) THEN
         FLYP=CC*VP
         ELSE

         ENDIF

         CI3=AN3(I3,J)
         ELSE

  191 CONTINUE

       SUBROUTINE ADVS()
                                    !DO I=1,IS1*JS1*KS1

                                    !ENDDO

      J1=J+1
      J2=J-1

!--------------------------------------------------

      I1=I+1
      I2=I-1
      DO 10 K=1,KI
!--------------------------------------------------
      CC= S2(I,J,K)
      CI1=S2(I1,J,K)
      CI2=S2(I2,J,K)
      CJ1=S2(I,J1,K)
      CJ2=S2(I,J2,K)
      UP=U2(I,J1,K)
      UU=0.5*(UP+U2(I,J,K))
      VP=V2(I1,J,K)
      VV=0.5*(VP+V2(I,J,K))
!-----------------------------------------------------
      A=ABS(UU)
      CDX=0.5*(UU+A)*ADX(I,J,K)*(CC-CJ2)&
	+0.5*(UU-A)*ADX(I,J1,K)*(CJ1-CC)
      A=ABS(VV)
      CDY=0.5*(VV+A)*ADY(I1,J,K)*(CC-CI1)&
	+0.5*(VV-A)*ADY(I,J,K)*(CI2-CC)
      TT(K)=S1(I,J,K)-C2*(CDX+CDY)
!------------------------------------------------------
      A=ABS(UP)
      FLXP=0.5*(UP+A)*CC+0.5*(UP-A)*CJ1
      APX(I,J1,K)=ADX(I,J1,K)*(UP*0.5*(CC+CJ1)-FLXP)*C2
      A=ABS(VP)       
      FLYP=0.5*(VP+A)*CC+0.5*(VP-A)*CI1
      APY(I+1,J,K)=ADY(I1,J,K)*(VP*0.5*(CC+CI1)-FLYP)*C2
!-------------------------------------------------------
   10 CONTINUE
!-------------------------------------------------------
      CK2=S2(I,J,1)
      CC=CK2
      WW1=W(I,J,1)
      DO K=1,KI-1   
      K1=K+1
      WW2=W(I,J,K1)
      WW=0.5*(WW1+WW2)
      A=ABS(WW)     
      CK1=S2(I,J,K1)
      CD(I,J,K)=TT(K)-DT*&
	(0.5*(WW+A)*(CC-CK2)/DZ(K)+0.5*(WW-A)*(CK1-CC)/DZ(K1))
      CK2=CC
      CC=CK1
      WW1=WW2 
      ENDDO  
      CD(I,J,KI)=TT(KI) 
      CC=S2(I,J,1)
      DO K=2,KI
      CK1=S2(I,J,K)
      WW=W(I,J,K)
      A=ABS(WW)
      APZ(I,J,K)=0.5*WW*(CC+CK1)-0.5*(WW+A)*CC+0.5*(WW-A)*CK1
      CC=CK1
      ENDDO
   11 CONTINUE
!********************************************************
      DO 91 J=1,JS
      DO 91 I=1,IS
      KI=IDX(I,J)
      IF(KI.EQ.0)GO TO 91
      J1=J+1
      J2=J-1
      DO K=1,KI
      A1=ADX(I,J1,K)*(CD(I,J1,K)-CD(I,J,K))
      B1=ADX(I,J2,K)*(CD(I,J2,K)-CD(I,J-2,K))
      A=APX(I,J,K)
      B=SIGN(1.,A)
      APX(I,J,K)=B*AMAX1(0.,AMIN1(ABS(A),B*A1,B*B1))
      ENDDO
   91 CONTINUE
!**********************************************************
      DO 191 J=1,JS
      J1=J+1
      DO 191 I=1,IS
      KI=KT1(I,J)
      IF(KI.EQ.0)GO TO 191
      DO K=1,KI
      CD(I,J,K)=CD(I,J,K)-(APX(I,J1,K)-APX(I,J,K))
      ENDDO
  191 CONTINUE
!******* y-COORDINATE
      DO 391 I=1,IS
      I1=I+1
      I2=I-1
      DO 391 J=1,JS
      KI=IDY(I,J)
      IF(KI.EQ.0)GO TO 391
      DO K=1,KI
      A1=ADY(I1,J,K)*(CD(I,J,K)-CD(I1,J,K))
      B1=ADY(I2,J,K)*(CD(I-2,J,K)-CD(I2,J,K))
      A=APY(I,J,K)
      B=SIGN(1.,A)
      APY(I,J,K)=B*AMAX1(0.,AMIN1(ABS(A),B*A1,B*B1))
      ENDDO
  391 CONTINUE

      DO 392 I=1,IS

      DO 392 J=1,JS
      KI=KT1(I,J)
      IF(KI.EQ.0)GO TO 392
      DO K=1,KI
      CD(I,J,K)=CD(I,J,K)-(APY(I,J,K)-APY(I1,J,K))
      ENDDO
  392 CONTINUE
!*******************************************************
      DO 491 I=1,IS
      DO 491 J=1,JS
      KI=KT1(I,J)
      IF(KI.EQ.0)GO TO 491
      B1=(CD(I,J,3)-CD(I,J,2))*DZ1(2)/DT
      A=APZ(I,J,2)
      B=SIGN(1.,A)
      APZ(I,J,2)=B*AMAX1(0.,AMIN1(ABS(A),B*B1))
      DO K=3,KI-1
      A1=(CD(I,J,K-1)-CD(I,J,K-2))*DZ1(K-1)/DT
      B1=(CD(I,J,K+1)-CD(I,J,K))*DZ1(K)/DT
      A=APZ(I,J,K)
      B=SIGN(1.,A)
      APZ(I,J,K)=B*AMAX1(0.,AMIN1(ABS(A),B*A1,B*B1))
      ENDDO
      A1=(CD(I,J,KI-1)-CD(I,J,KI-2))*DZ1(KI-1)/DT
      A=APZ(I,J,KI)
      B=SIGN(1.,A)
      APZ(I,J,KI)=B*AMAX1(0.,AMIN1(ABS(A),B*A1))
  491 CONTINUE
!**********************************************************
      DO 591 I=1,IS
      DO 591 J=1,JS
      KI=KT1(I,J)
      IF(KI.EQ.0)GO TO 591
      DO K=2,KI-1
      CD(I,J,K)=CD(I,J,K)-(APZ(I,J,K+1)-APZ(I,J,K))*DT/DZ1(K)
      ENDDO
  591 CONTINUE
!
                                          !DO I=1,IJKS
      S2=CD                               !S2(I,1,1)=CD(I,1,1)
                                          !ENDDO

      B=(SXX(I,J)+SYY(I,J))*0.5
      IF(A.GT.0..AND.B.GE.0.) GOTO 3
!
      SSXX=EXX(I,J)*1.E7
      SSYY=EYY(I,J)*1.E7
      SSXY=EXY(I,J)*1.E7
      A=(SSXX+SSYY)*0.5
      B1=(SSXX-SSYY)*0.5
      B2=SQRT(B1*B1+SSXY*SSXY)
      A1=A+B2
      A2=A-B2
!
      DO K=2,NGR
      IF(AN1(I,J,K+1).GT.0.05) GOTO 5
      ENDDO
    5 SCRIT=-0.43E5*HST(K)**2
      IF(A2.GT.SCRIT.AND.A1.LT..0) GOTO 4
      IF(A2.LT.SCRIT) A2=SCRIT
      IF(A1.LT.SCRIT) A1=SCRIT
      IF(A1.GT.0.) A1=0.
      IF(A2.GT.0.) A2=0.       !!!!!!!!!!!!!!!!
      ANGL=ATAN2(SSXY,B1)*0.5
      COR=COS(-ANGL)
      SOR=SIN(-ANGL)
      SXX(I,J)=A1*COR*COR+A2*SOR*SOR
      SYY(I,J)=A1*SOR*SOR+A2*COR*COR
      SXY(I,J)=-(A1-A2)*COR*SOR
      EXX(I,J)=SXX(I,J)*1.E-7
      EYY(I,J)=SYY(I,J)*1.E-7
      EXY(I,J)=SXY(I,J)*1.E-7
      GOTO 1
    3 EXX(I,J)=0.
      EYY(I,J)=0.
      EXY(I,J)=0.
      SXX(I,J)=0.
      SYY(I,J)=0.
      SXY(I,J)=0.
      GOTO 1
 4    SXX(I,J)=SSXX
      SYY(I,J)=SSYY
      SXY(I,J)=SSXY
    1 CONTINUE
!
      RETURN
      END subroutine
	  end module

      OPEN(1,FILE='FI1DL1.DAT')

      DO 50 I=1,135

      Y0=Y01(NOM)

      DO K=1,4

      an4_wind=AB*(1.-DX)*(1.+DY)

      DO 60 I=1,IS

      VY= PX*BLL

      AU=(30.-0.8333*V_wind)

!===================================================

      SUBROUTINE SURFW()
!
      OPEN(17,FILE=EWIN)
!-----------------------------------------------------------------
        NC=5            !  Интервал между стрелками в шагах сетки
!-----------------------!-----------------------------------------
        FT=0.5          !  Масштаб стрелок в шагах сетки
!------------------------------------------------------------------
        NI=IS1
!
        UMAX=0.
        I1=-NC+1
        NNN=5
        DO 735 I=1,IS1,NC
        I1=I1+NC
        J1=-NC+1
        Y11=NI+1-I1
        DO 734 J=1,JS1,NC
        J1=J1+NC
        IF(HT(I,J).EQ.8888.)GO TO 734
        PPP=WIND(I,J)
        IF(PPP.GT.UMAX)UMAX=PPP
        PI1=ALF(I,J)/57.3
        X1=J1
        SA=PPP/4.*FT
        X2=PPP*SIN(PI1)*FT+X1
        Y22=PPP*COS(PI1)*FT+Y11
        X3=X2+SA*SIN(PI1+165./57.3)
        Y3=Y22+SA*COS(PI1+165./57.3)
        X4=X2+SA*SIN(PI1+195./57.3)
        Y4=Y22+SA*COS(PI1+195./57.3)
        WRITE(17,110)NNN
        WRITE(17,109)X1,Y11
        WRITE(17,109)X2,Y22
        WRITE(17,109)X3,Y3
        WRITE(17,109)X2,Y22
        WRITE(17,109)X4,Y4
 734    CONTINUE
 735    CONTINUE
        CLOSE(17)
!
 109    FORMAT(2F7.2)
 110    FORMAT(I6)
!
        WRITE(*,*)UMAX
        WRITE(15,*)UMAX
!
        RETURN
        END subroutine
