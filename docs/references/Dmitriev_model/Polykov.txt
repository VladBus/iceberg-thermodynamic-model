      SUBROUTINE ADV2D(DT,C2)
      include 'comblk'
C
      DO I=1,IJS1
      APX2(I,1)=0.
      APY2(I,1)=0.
      ENDDO
C****************************************************
      DO 11 I=1,IS
      I1=I+1 
      I2=I-1
      I3=I+2
      DO 11 J=1,JS
C----------------------------------------------------
      KI=KT1(I,J)*IT(I,J)*(1.-PRIP(I,J))
      IF(KI.EQ.0) GOTO 11
      J1=J+1
      J2=J-1
      UP=50.*(UIC2(I,J1)+UIC2(I1,J1))
      UM=50.*(UIC2(I,J)+UIC2(I1,J))
      UU=0.5*(UP+UM)
      VP=50.*(VIC2(I1,J)+VIC2(I1,J1))
      VM=50.*(VIC2(I,J)+VIC2(I,J1))
      VV=0.5*(VP+VM)
C-----------------------------------------------------
      CC=AN4(I,J)
      CI1=AN4(I1,J)
      CI2=AN4(I2,J)
      CJ1=AN4(I,J1)
      CJ2=AN4(I,J2)
C------------------------------------------------------
      a2=abs(UP)
      a=0.5*((UP+a2)*CC+(UP-a2)*CJ1)
      a2=abs(UM)
      b=0.5*((UM+a2)*CJ2+(UM-a2)*CC)
      a2=abs(VM)
      a1=0.5*((VM+a2)*CC+(VM-a2)*CI2)
      a2=abs(VP)
      b1=0.5*((VP+a2)*CI1+(VP-a2)*CC)     
      CD2(I,J)=AN5(I,J)-C2*(a-b+a1-b1)
C-------------------------------------------------------
      A=ABS(UP)
      FLXP=0.5*(UP+A)*CC+0.5*(UP-A)*CJ1
      APX2(I,J1)=AIDX(I,J1)*
     *(UP*(7.*(CC+CJ1)-(CJ2+AN4(I,J+2)))/12.-FLXP)*C2
      A=ABS(VP)
      FLYP=0.5*(VP+A)*CI1+0.5*(VP-A)*CC
      APY2(I1,J)=AIDY(I1,J)*
     *(VP*(7.*(CC+CI1)-(CI2+AN4(I3,J)))/12.-FLYP)*C2
   11 CONTINUE
C********************************************************
      DO 91 I=1,IS
      DO 91 J=1,JS
      IF(IDX(I,J).EQ.0) GOTO 91
      J1=J+1
      J2=J-1
      A1=AIDX(I,J1)*(CD2(I,J1)-CD2(I,J))
      B1=AIDX(I,J2)*(CD2(I,J2)-CD2(I,J-2))
      A=APX2(I,J)
      B=SIGN(1.,A)
      APX2(I,J)=B*AMAX1(0.,AMIN1(ABS(A),B*A1,B*B1))
   91 CONTINUE
C**********************************************************
      DO 191 J=1,JS
      J1=J+1
      DO 191 I=1,IS
      KI=KT1(I,J)*IT(I,J)*(1.-PRIP(I,J))
      IF(KI.NE.0) CD2(I,J)=CD2(I,J)-(APX2(I,J1)-APX2(I,J))
  191 CONTINUE
C******* y-COORDINATE
      DO 391 I=1,IS
      I1=I+1
      I2=I+2
      DO 391 J=1,JS
      IF(IDY(I,J).EQ.0) GOTO 391
      A1=AIDY(I1,J)*(CD2(I,J)-CD2(I1,J))
      B1=AIDY(I2,J)*(CD2(I-2,J)-CD2(I2,J))
      A=APY2(I,J)
      B=SIGN(1.,A)
      APY2(I,J)=B*AMAX1(0.,AMIN1(ABS(A),B*A1,B*B1))
  391 CONTINUE
C**********************************************************
      DO 392 I=1,IS
      I1=I+1
      DO 392 J=1,JS
      KI=KT1(I,J)*IT(I,J)*(1.-PRIP(I,J))
      IF(KI.NE.0) AN4(I,J)=CD2(I,J)-(APY2(I,J)-APY2(I1,J))
  392 CONTINUE
C*******************************************************
      DO I=1,IS
      I1=I+1
      I2=I-1 
      DO J=1,JS
      IF(IT(I,J).EQ.0) AN4(I,J)=
     *0.5*(AN4(I,J-1)+AN4(I,J+1)+AN4(I1,J)+AN4(I2,J))
      ENDDO
      ENDDO
C*******************************************************
      DO I=1,IS
      AN4(I,1)=AN4(I,2)
      AN4(I,JS)=AN4(I,JS2)
      ENDDO
      DO J=1,JS
      AN4(1,J)=AN4(2,J)
      AN4(IS,J)=AN4(IS2,J)
      ENDDO
C*******************************************************
      RETURN
      END


      SUBROUTINE ADVS(DT,C2)
      INCLUDE 'comblk'
C**************************************************
      DO I=1,IS1*JS1*KS1
      APZ(I,1,1)=0.
      ENDDO
      DO I=1,IJKS
      APX(I,1,1)=0.
      APY(I,1,1)=0.
      ENDDO
C
      DO 11 J=1,JS
      J1=J+1
      J2=J-1
      DO 11 I=1,IS
C--------------------------------------------------
      KI=KT1(I,J)
      IF(KI.EQ.0) GOTO 11
      I1=I+1
      I2=I-1
      DO 10 K=1,KI
C--------------------------------------------------
      CC= S2(I,J,K)
      CI1=S2(I1,J,K)
      CI2=S2(I2,J,K)
      CJ1=S2(I,J1,K)
      CJ2=S2(I,J2,K)
      UP=U2(I,J1,K)
      UU=0.5*(UP+U2(I,J,K))
      VP=V2(I1,J,K)
      VV=0.5*(VP+V2(I,J,K))
C-----------------------------------------------------
      A=ABS(UU)
      CDX=0.5*(UU+A)*ADX(I,J,K)*(CC-CJ2)
     *+0.5*(UU-A)*ADX(I,J1,K)*(CJ1-CC)
      A=ABS(VV)
      CDY=0.5*(VV+A)*ADY(I1,J,K)*(CC-CI1)
     *+0.5*(VV-A)*ADY(I,J,K)*(CI2-CC)
      TT(K)=S1(I,J,K)-C2*(CDX+CDY)
C------------------------------------------------------
      A=ABS(UP)
      FLXP=0.5*(UP+A)*CC+0.5*(UP-A)*CJ1
      APX(I,J1,K)=ADX(I,J1,K)*(UP*0.5*(CC+CJ1)-FLXP)*C2
      A=ABS(VP)       
      FLYP=0.5*(VP+A)*CC+0.5*(VP-A)*CI1
      APY(I+1,J,K)=ADY(I1,J,K)*(VP*0.5*(CC+CI1)-FLYP)*C2
C-------------------------------------------------------
   10 CONTINUE
C-------------------------------------------------------
      CK2=S2(I,J,1)
      CC=CK2
      WW1=W(I,J,1)
      DO K=1,KI-1   
      K1=K+1
      WW2=W(I,J,K1)
      WW=0.5*(WW1+WW2)
      A=ABS(WW)     
      CK1=S2(I,J,K1)
      CD(I,J,K)=TT(K)-DT*
     *(0.5*(WW+A)*(CC-CK2)/DZ(K)+0.5*(WW-A)*(CK1-CC)/DZ(K1))
      CK2=CC
      CC=CK1
      WW1=WW2 
      ENDDO  
      CD(I,J,KI)=TT(KI) 
      CC=S2(I,J,1)
      DO K=2,KI
      CK1=S2(I,J,K)
      WW=W(I,J,K)
      A=ABS(WW)
      APZ(I,J,K)=0.5*WW*(CC+CK1)-0.5*(WW+A)*CC+0.5*(WW-A)*CK1
      CC=CK1
      ENDDO
   11 CONTINUE
C********************************************************
      DO 91 J=1,JS
      DO 91 I=1,IS
      J1=J+1
      J2=J-1
      DO K=1,IDX(I,J)
      A1=ADX(I,J1,K)*(CD(I,J1,K)-CD(I,J,K))
      B1=ADX(I,J2,K)*(CD(I,J2,K)-CD(I,J-2,K))
      A=APX(I,J,K)
      B=SIGN(1.,A)
      APX(I,J,K)=B*AMAX1(0.,AMIN1(ABS(A),B*A1,B*B1))
      ENDDO
   91 CONTINUE
C**********************************************************
      DO 191 J=1,JS
      J1=J+1
      DO 191 I=1,IS
      DO K=1,KT1(I,J)
      CD(I,J,K)=CD(I,J,K)-(APX(I,J1,K)-APX(I,J,K))
      ENDDO
  191 CONTINUE
C******* y-COORDINATE
      DO 391 I=1,IS
      I1=I+1
      I2=I-1
      DO 391 J=1,JS
      DO K=1,IDY(I,J)
      A1=ADY(I1,J,K)*(CD(I,J,K)-CD(I1,J,K))
      B1=ADY(I2,J,K)*(CD(I-2,J,K)-CD(I2,J,K))
      A=APY(I,J,K)
      B=SIGN(1.,A)
      APY(I,J,K)=B*AMAX1(0.,AMIN1(ABS(A),B*A1,B*B1))
      ENDDO
  391 CONTINUE
C**********************************************************
      DO 392 I=1,IS 
      I1=I+1
      DO 392 J=1,JS
      DO K=1,KT1(I,J)
      CD(I,J,K)=CD(I,J,K)-(APY(I,J,K)-APY(I1,J,K))
      ENDDO
  392 CONTINUE
C*******************************************************
      DO 491 I=1,IS
      DO 491 J=1,JS
      KI=KT1(I,J)
      B1=(CD(I,J,3)-CD(I,J,2))*DZ1(2)/DT
      A=APZ(I,J,2)
      B=SIGN(1.,A)
      APZ(I,J,2)=B*AMAX1(0.,AMIN1(ABS(A),B*B1))
      DO K=3,KI-1
      A1=(CD(I,J,K-1)-CD(I,J,K-2))*DZ1(K-1)/DT
      B1=(CD(I,J,K+1)-CD(I,J,K))*DZ1(K)/DT
      A=APZ(I,J,K)
      B=SIGN(1.,A)
      APZ(I,J,K)=B*AMAX1(0.,AMIN1(ABS(A),B*A1,B*B1))
      ENDDO
      A1=(CD(I,J,KI-1)-CD(I,J,KI-2))*DZ1(KI-1)/DT
      A=APZ(I,J,KI)
      B=SIGN(1.,A)
      APZ(I,J,KI)=B*AMAX1(0.,AMIN1(ABS(A),B*A1))
  491 CONTINUE
C**********************************************************
      DO 591 I=1,IS
      DO 591 J=1,JS
      DO K=2,KT1(I,J)-1
      CD(I,J,K)=CD(I,J,K)-(APZ(I,J,K+1)-APZ(I,J,K))*DT/DZ1(K)
      ENDDO
  591 CONTINUE
C
      DO I=1,IJKS
      S3(I,1,1)=CD(I,1,1)
      ENDDO
C*******************************************************
      RETURN
      END

      subroutine HEAT(KCT,NDAY,DTA,LLL)
      INCLUDE 'comblk'
C
      DZZ=DZ1(1)/100.
      DTZ=DTA/(DZZ*4.19E6)
      DO 1 I=1,IS
      DO 1 J=1,JS
      KI=KT1(I,J)*IFIX(1.-PRIP(I,J))
      IF(KI.EQ.0) GOTO 1
C calculation of solar radiation
      TTAC=TATM(I,J)                    ! air temperature (C)
      TTA=TTAC+273.15                   ! air temperature (K)
      PPATM=PATM(I,J)*100.              ! atm.pressure (Pask)
      RATM=PPATM/287./TTA               ! air density
      TWAC=T1(I,J,1)                    ! water temperature (C)
      TWA=TWAC+273.15                   ! water temperature (K)
      HHUM=HUMID(I,J)                   ! humidity (parts of unit)
      WW=WIND(I,J)                      ! geostr. wind  (m/s)
      CCLO=CLOUD(I,J)                   ! cloudiness
      A=S1(I,J,1)
      DO K=1,NGR
      ANP(K)=AN1(I,J,K+1)
      IF(ANP(K).NE.0.) THEN
      HICP(K)=WICE1(I,J,K)/ANP(K)
      HSNP(K)=HSNOW(I,J,K)
      ELSE
      HICP(K)=0.
      HSNP(K)=0.
      ENDIF
      ENDDO
      DO K=1,NGR1
      SPAR(K)=A
      TPAR(K)=TWA
      ENDDO
      ANSUM=ANS(I,J)
      TFR=-54.*SPAR(1)                  ! freezing temperature (C)
      FW=1028.*4.1868E3*0.01*SKT(I,J)*(T1(I,J,1)-TFR)/DZ(1)
c      fw=2. !!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!
      FW=0.5*(FW+ABS(FW))
      FW1(i,j)=fw
      TFR=TFR+273.15                    ! freezing temperature (K)
      A1=FI(I,J)/57.3                   ! constants for solar radiation
      A2=SIN(A1)
      B2=COS(A1)
      B=23.44*COS((172-NDAY)/57.3)/57.3
      A3=SIN(B)
      B3=COS(B)
      CZ=0.
C      DO K=III-7,III       ! THAT IS FOR 1-HOUR TIME STEP DT
c      B1=A2*A3+B2*B3*COS((12-K)*0.2617994) ! THAT IS AS USUAL
      B1=A2*A3+B2*B3*COS((12-(KCT*8-4))*0.2617994)
C      IF(B1.LT.0.) B1=0.
      B1=0.5*(B1+ABS(B1))
      CZ=CZ+B1
C      ENDDO
C      CZ=CZ/8.              ! THAT IS FOR 1-HOUR TIME STEP DT
      A1=1353.*CZ*CZ*(1.-0.6*CCLO*CCLO*CCLO)
      B1=(CZ+2.7)*1.E-5
      B2=1.085*CZ+0.1
      E=HHUM*610.78*10**(8.61503*(TTA-273.15)/TTA) ! vapor pressure
      SW=A1/(B1*E+B2)                  ! solar radiation (f-lae 1.4,1.5)
      B3=RATM*1.7068*WW                ! constant for SH
      A3=0.6650735*RATM/PPATM*WW       ! constant for EL
      WL=5.4999E-8*TTA**4*(1.-0.261*EXP
     *(-7.77E-4*((273.15-TTA)**2)))*
     *(1.+0.275*CCLO)                  ! incoming longwave rad.F (f-la 1.3)
C+++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++
C     fluxes and possible formation of ice (no ice is initial)
C+++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++
      HFIRST=0.
      ANN1=AN1(I,J,1)
      IF(ANN1.EQ.0.) GOTO 11
      SH1=B3*(TTA-TWA)                            ! SH (f-la 1.2)
      EL1=A3*2.5E6*(HHUM*10**(7.63*TTAC/(241.9+TTAC))-
     *10**(7.63*TWAC/(241.9+TWAC)))               ! LE (f-la 1.1)
      AAA=-0.97*5.67E-8*(TWA**4)
      QN=SH1+EL1+0.97*WL+0.9*SW-0.97*5.67E-8*(TWA**4)
C
      IF(QN.GT.0.) THEN
      TPAR(1)=TPAR(1)+DTZ*ANN1*QN                 ! ANN1 - for leads
      ELSE
      TPAR(1)=TPAR(1)+DTZ*QN
      ENDIF
      IF(TPAR(1).GE.TFR) THEN
      IF(TPAR(1).GT.TTA.AND.QN.GT.0.) TPAR(1)=TTA ! no ice formation
      ELSE
      TPAR(1)=TFR                                 !  ice formation
      HFIRST=0.01
      ENDIF
C
   11 CONTINUE
C@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@
C                     fluxes with ice
C@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@
      DO 20 K=1,NGR
      K1=K+1
      IF(ANP(K).LT.0.001) GOTO 20
C
      DHSN=0.
      DHIC1=0.
      TTS=0.5*(TTA+TWA)
      EL=A3*2.834E6*(HHUM*10**(9.5*TTAC/(265.5+TTAC))-
     *10**(9.5*(TTS-273.15)/(TTS-7.65)))      ! EL (f-la 1.1)
      HHIC1=1./HICP(K)
      IF(HSNP(K).NE.0.) GOTO 30
C****************** ice without snow    *******************
      B=2.04*HHIC1
      ERR=100.
      DO WHILE(ERR.GE.0.25)
      SH=B3*(TTA-TTS)                         ! SH (f-la 1.2)
      A=5.4999E-8*TTS**3                      ! 5.4999E-8=(0.97*5.67E-8)
      A1=SH+EL+0.466*SW+0.97*WL
      DTTS=(A1-A*TTS+B*(TFR-TTS))/(4.*A+B)
      A=TTS+DTTS                              ! new surface temperature
      ERR1=ERR
      ERR=ABS(DTTS)
      IF(ERR.GE.ERR1) GOTO 40
      TTS=A
      ENDDO
      GOTO 41
   40 TTS=ALFA(LLL)*TTA+(1.-ALFA(LLL))*273.15
      SH=B3*(TTA-TTS)  
      A1=SH+EL+0.466*SW+0.97*WL       
   41 CONTINUE
C****  processes at the surface "ice-air"
      IF(TTS.LE.273.15) GOTO 25
C    melting at the ice surface
      TTS=273.15
      DHIC1=-DTA/302.E6*(A1-5.4999E-8*TTS**4+B*(TFR-TTS))
      HICP(K)=HICP(K)+DHIC1
      IF(HICP(K).LT.0.01) HICP(K)=0.
      GOTO 26
C    snow fall parameterization
   25 DHSN=DTA*SFAL(LLL)
      HSNP(K)=DHSN
C****  processes at the surface "ice-water"
   26 DHIC=-DTA/302.E6*(FW-B*(TFR-TTS))
      HICP(K)=HICP(K)+DHIC
      DHIC=DHIC+DHIC1
      IF(HICP(K).LT.0.01) THEN
      HICP(K)=0.
      HSNP(K)=0.
      ENDIF
      GOTO 21
C******************  ice with snow ************************
   30 CONTINUE
      B1=0.31*HICP(K)
      B2=2.04*HSNP(K)
      B4=B1+B2
      B=0.6324/B4                           ! 0.6324=2.04*0.31
      ERR=100.
      DO WHILE(ERR.GE.0.25)
      SH=B3*(TTA-TTS)                       ! SH (f-la 1.2)
      A=5.6133E-8*TTS**3                    ! 5.6133E-8=(0.99*5.67E-8)
      A1=SH+EL+(1.-ALSN(LLL))*SW+0.99*WL
      DTTS=(A1-A*TTS+B*(TFR-TTS))/(4.*A+B)
      A=TTS+DTTS                            ! new snow Tsur
      ERR1=ERR
      ERR=ABS(DTTS)
      IF(ERR.GE.ERR1) GOTO 45
      TTS=A
      A1=EL+(1.-ALSN(LLL))*SW+0.99*WL
      ENDDO
      GOTO 46
   45 TTS=ALFA(LLL)*TTA+(1.-ALFA(LLL))*273.15
      SH=B3*(TTA-TTS)  
      A1=SH+EL+0.466*SW+0.97*WL  
   46 CONTINUE
C**** processes at the surface "snow-air"
      IF(TTS.LE.273.15) THEN
      DHSN=SFAL(LLL)*DTA                             ! snow fall
      TTI=(B1*TTS+B2*TFR)/B4                        ! new ice Tsfc
      ELSE
      TTS=273.15                                    !snow melting
      TTI=(B1*273.15+B2*TFR)/B4                     ! new ice Tsfc
      DHSN=-DTA/110.E6*(A1-5.6133E-8*TTS**4+
     *0.31/HSNP(K)*(TTI-TTS))                       ! f-la (2.14)
      ENDIF
C**** processes at the surface "ice-water"
      DHIC=DTA/302.E6*(2.04*(TFR-TTI)/HICP(K)-FW)    ! f-la (2.15) !!!!!!!!!
      HICP(K)=HICP(K)+DHIC
      IF(HICP(K).GE.0.01) THEN
      A=0.1*HICP(K)
      HSNP(K)=HSNP(K)+DHSN
      IF(HSNP(K).GT.A) THEN
      HSNP(K)=A
      ELSE
      IF(HSNP(K).LT.0.) HSNP(K)=0.
      ENDIF
      ELSE
      HICP(K)=0.
      HSNP(K)=0.
      ENDIF
C********* new water temperature and salinity values *********
   21 A=0.88*DHIC
      B=DZZ-A
      IF(DHIC.LT.0.) THEN
         IF(DHSN.LT.0.) THEN                          ! melting of
         A1=0.3*DHSN                                  ! ice + snow
         B=B-A1
         B1=273.15*A1
         ELSE
         B1=0.
         A1=0.
         ENDIF
         SPAR(K1)=(SPAR(K1)*(DZZ+A+A1)-SICST(K)*A)/DZZ   ! when melting
         TFR=-54.*SPAR(K1)+273.15
         TPAR(K1)=(TFR*DZZ-B1)/(DZZ-A1)                  ! when melting
      ELSE
         SPAR(K1)=(SPAR(K1)*DZZ-SICST(K)*A)/B         ! when freezing
         TFR=-54.*SPAR(K1)+273.15
         TPAR(K1)=TFR
      ENDIF
      HSNOW(I,J,K)=HSNP(K)
      HICE(I,J,K)=HICP(K)
      IF(HICP(K).EQ.0.) THEN
      ANN1=ANN1+ANP(K)
      AN1(I,J,K1)=0.
      ANP(K)=0.
      ENDIF
   20 CONTINUE
C
      IF(ANN1.GT.0..AND.ANN1.LT.1..AND.QN.GT.0.) THEN
C************ decreasing of ice area due to melting in leads *******
         A=ANN1*QN*DTA
         B=0.
         A3=0.
         B3=0.
         DO 55 K=1,NGR
         IF(HICP(K).NE.0.) THEN
         K1=K+1
         DANP(K)=ANP(K)*A/(302.E6*HICP(K)+110.E6*HSNP(K))   ! f-la 3.1
         A5=ANP(K)-DANP(K)
         DANP(K)=DANP(K)+0.5*(A5-ABS(A5))
         ANP(K)=ANP(K)-DANP(K)                              ! f-la 3.2
         B=B+DANP(K)
         A3=A3+ANP(K)*TPAR(K1)
         B3=B3+ANP(K)*SPAR(K1)
         ELSE
         DANP(K)=0.
         ENDIF
   55    CONTINUE
         ANN2=ANN1+B                                        ! f-la 3.3
         A2=0.
         B2=0.
         DO 60 K=1,NGR
         K1=K+1
         TFR=-54.*SICST(K)+273.15                           ! f-la 3.5
         A=0.88*HICP(K)
         A1=DANP(K)/(A+DZZ)
         A2=A2+(TPAR(K1)*DZZ+TFR*A)*A1                      ! f-la 3.4
         B2=B2+(SPAR(K1)*DZZ+SICST(K)*A)*A1                ! f-la 3.4
   60    CONTINUE
         TPAR(1)=(TPAR(1)*ANN1+A2)/ANN2                     ! f-la 3.6
         SPAR(1)=(SPAR(1)*ANN1+B2)/ANN2                     ! f-la 3.6
         AN1(I,J,1)=ANN2
         T1(I,J,1)=ANN2*TPAR(1)+A3-273.15                   ! f-la 3.7
         S1(I,J,1)=ANN2*SPAR(1)+B3                          ! f-la 3.7
      ELSE
C****  new water temperature and salinity without lateral melting ***
         A1=ANN1*TPAR(1)
         B1=ANN1*SPAR(1)
         DO K=1,NGR
         K1=K+1
         A1=A1+ANP(K)*TPAR(K1)
         B1=B1+ANP(K)*SPAR(K1)
         ENDDO
         T1(I,J,1)=A1-273.15
         S1(I,J,1)=B1
         IF(HFIRST.GT.0.) THEN
         A2=AN1(I,J,2)
         A1=ANN1+A2
         HICP(1)=(0.01*ANN1+HICP(1)*A2)/A1
         AN1(I,J,1)=0.
         ANP(1)=A1
         ENDIF
         ENDIF
C      AN1(I,J,1)=ANN1
      DO K=1,NGR
      HICE(I,J,K)=HICP(K)
      HSNOW(I,J,K)=HSNP(K)
      AN1(I,J,K+1)=ANP(K)
      WICE1(I,J,K)=HICE(I,J,K)*ANP(K)
      ENDDO
C
    1 CONTINUE
C
      RETURN
      END


C     Attention: this code is for leapfrog scheme. Change lines
C                 128-131 for Euler scheme
C 
      SUBROUTINE kash1
      INCLUDE 'comblk'
c
      open(1,file='kash-bot')
      j1=-14
      j2=0
      do kk=1,4
      j1=j1+15
      j2=j2+15
      do i=1,31
      read(1,50) (ht(i,j),j=j1,j2)
 50   format(4x,15f5.0) 
      enddo
      read(1,50)
      enddo

      j2=j2+6
      do i=1,31
      read(1,51) (ht(i,j),j=j1,j2)
 51   format(4x,6f5.0) 
      enddo      
      close(1)

      do i=1,ijs

      if(ht(i,1).ne.8888.) then
      ht(i,1) = ht(i,1)*100.
c      ht(i,1)=10000.       !!!!!!!!!!!!!!!!!!!!!!!!!!!!
      endif
      enddo
      do i=1,is1
      ht(i,js1)=8888.
      enddo
      do j=1,js1
      ht(is1,j)=8888. 
      enddo  
c    
      do j=2,js
      do i=1,is
      a=ht(i,j)
      b=ht(i,j-1)
      if(a.ne.8888..and.b.ne.8888.) hu(i,j)=0.5*(a+b)
      enddo
      enddo
      do j=1,js
      do i=2,is
      a=ht(i,j)
      b=ht(i+1,j)
      if(a.ne.8888..and.b.ne.8888.) hv(i,j)=0.5*(a+b)
      enddo
      enddo  
      do j=1,js
      hv(is,j)=ht(is,j)
      enddo
C***** calculation of DZ, DZ1, KT1 arrays (point T,S)
      DO K=1,KS
      Z(K)=Z(K)*100.
      ENDDO
C 
      DO 34 K=1,KS
      DZ1(K)=8888.
   34 DZ(K)=8888.
      DZ(KS1)=8888.
      DZ1(1)=0.5*(Z(2)+Z(1))
      DO 36 K=2,KS2
      DZ1(K)=0.5*(Z(K+1)-Z(K-1))
   36 CONTINUE
      DZ1(KS)=0.5*(Z(KS)-Z(KS2))
      DZ(1)=Z(1)
      DO 41 K=2,KS
      DZ(K)=Z(K)-Z(K-1)
   41 CONTINUE
      DO 43 I=1,IS1
      DO 43 J=1,JS1
      KT1(I,J)=0
      HHT=HT(I,J)
      hhu = hu(i,j)
      hhv = hv(i,j)
      IF(HHT.EQ.8888.) GOTO 63
      KT1(I,J)=1
      DO 44 K=2,KS
      IF(HHT.LT.Z(K)) GOTO 63
   44 KT1(I,J)=K
   63 CONTINUE
      IF(hhu.EQ.8888.) GOTO 73
      ku(i,j) = 1
      DO k = 2,ks
      IF(hhu.LT.z(k)) GO TO 73
      ku(i,j) = k
      ENDDO
 73   CONTINUE
      IF(hhv.EQ.8888.) GOTO 43
      kv(i,j) = 1
      DO k=2,ks
       IF(hhv.LT.z(k)) GO TO 43
       kv(i,j) = k
      ENDDO
  43  CONTINUE
      DO i=1,ijks
      uro(i,1,1)=0.
      vro(i,1,1)=0.
      ENDDO
      DO I=1,IS
      DO J=2,JS
      if(KT1(i,j).gt.KT1(i,j-1)) then
      do k=1,KT1(i,j-1)
      uro(i,j,k)=1.
      enddo
      else
      do k=1,KT1(i,j)
      uro(i,j,k)=1.
      enddo
      endif
      enddo
      enddo
      DO I=2,IS
      DO J=1,JS
      if(KT1(i,j).gt.KT1(i-1,j)) then
      Do k=1,KT1(i-1,j)
      vro(i,j,k)=1.
      enddo
      else
      do k=1,kt1(i,j)
      vro(i,j,k)=1.
      enddo
      endif
      enddo
      enddo
        DO 82   i=1,is1
          DO 82 j=1,js1
            IF(hu(i,j).EQ.8888.) GO TO 83
            ki=ku(i,j)
            a=hu(i,j)-z(ki) 
            if(a.lt.50.) a=50.           
            cfu(i,j)=0.16/LOG(a)**2
  83        CONTINUE
            IF(hv(i,j).EQ.8888.) GO TO 82
            ki=kv(i,j)
            a=hv(i,j)-z(ki)
            if(a.lt.50.) a=50.
            cfv(i,j)=0.16/LOG(a)**2
  82    CONTINUE
C ************************ K K 1 (ice point) *********************
      DO I=1,IS1
      DO J=1,JS1
      KK1(I,J)=0
      ENDDO
      ENDDO
      DO 24 I=2,IS
      I2=I-1
      DO 24 J=2,JS
      J2=J-1
      IF(KT1(I,J).EQ.0.OR.KT1(I,J2).EQ.0.OR.KT1(I2,J).EQ.0.
     *OR.KT1(I2,J2).EQ.0) GOTO 24
      KK1(I,J)=1
   24 CONTINUE
C********************* CORIOLIS PARAMETERS ***************
      A=2.*7.29E-5
      B=76.5+1./12.
      DO I=1,IS1
      B=B-1./12.
      DO J=1,JS1
      FU(I,J)=A*SIN(B/57.3)
      FV(I,J)=A*SIN(B/57.3)
      ENDDO
      ENDDO
      DO I=2,IS1
      I2=I-1
      DO J=2,JS1
      FKU(I,J)=0.25*(FU(I,J)+FU(I2,J)+FV(I,J)+FV(I,J-1))
      ENDDO
      ENDDO
      DO I=1,IS1
      FKU(I,1)=FV(I,1)
      ENDDO
      DO J=1,JS1
      FKU(1,J)=FU(1,J)
      ENDDO
C
      DO k=2,ks-1
        c31(k)=1./dz1(k)/dz(k+1)
        c34(k)=1./dz1(k)/dz(k)
      ENDDO
C
      DO i=1,ijks
      sku(i,1,1)=0
      skv(i,1,1)=0
      ENDDO
      DO 100 i = 1,is1   !
      i1=i+1
      DO 100 j = 1,js1
      ki=ku(i,j)
      if(ki.eq.0) goto 100
      j2=j-1
      kv1 = kv(i,j2)
      kv2 = kv(i,j)
      kv3 = kv(i1,j)
      kv4 =kv(i1,j2)
      DO k=1,ki
        kx = 0
        IF(k.LE.kv1)kx=kx+1
        IF(k.LE.kv2)kx=kx+1
        IF(k.LE.kv3)kx=kx+1
        IF(k.LE.kv4)kx=kx+1
      if(kx.eq.0) then
      sku(i,j,k)=0
      else
      sku(i,j,k)=1./float(kx)
      endif 
       ENDDO
  100 continue
c
      do 101 i=1,is1
      i2=i-1
      do 101 j=1,js1
      ki=kv(i,j)
      if(ki.eq.0) goto 101
      j1=j+1
        ku1 = ku(i2,j)
        ku2 = ku(i2,j1)
        ku3 = ku(i,j1)
        ku4 =ku(i,j)
C
       DO k=1,ki
        ky=0
        IF(k.LE.ku1)ky=ky+1
        IF(k.LE.ku2)ky=ky+1
        IF(k.LE.ku3)ky=ky+1
        IF(k.LE.ku4)ky=ky+1
        if(ky.eq.0) then
        skv(i,j,k)=0
        else
        skv(i,j,k)=1./float(ky)
       endif
       ENDDO
  101 continue
C***** calculation of IDX & IDY arrays (between T&S points)
      DO I=1,IJS
      IDY(I,1)=0
      IDX(I,1)=0
      ENDDO
      DO I=1,IS
      DO J=2,JS
      IF(KT1(I,J).LT.KT1(I,J-1)) THEN
      IDX(I,J)=KT1(I,J)
      ELSE
      IDX(I,J)=KT1(I,J-1)
      ENDIF
      ENDDO
      ENDDO
      DO I=2,IS
      DO J=1,JS
      IF(KT1(I,J).LT.KT1(I-1,J)) THEN
      IDY(I,J)=KT1(I,J)
      ELSE
      IDY(I,J)=KT1(I-1,J)
      ENDIF
      ENDDO
      ENDDO
C
      DO I=1,IJKS
      ADX(I,1,1)=0.
      ADY(I,1,1)=0.
      ENDDO
      DO I=2,IS
      DO J=2,JS
      DO K=1,IDX(I,J)
      ADX(I,J,K)=1.
      ENDDO
      DO K=1,IDY(I,J)
      ADY(I,J,K)=1.
      ENDDO
      ENDDO
      ENDDO   
C**** calculation of IT array (identificator for the ice adv.)
      DO I=1,IJS
      IT(I,1)=1
      ENDDO
      DO 200 I=1,IS
      I1=I+1
      DO 200 J=1,JS
      IF(KT1(I,J).EQ.0) GOTO 200
      J1=J+1
      IX1=IDX(I,J)
      IX2=IDX(I,J1)
      IY1=IDY(I,J)
      IY2=IDY(I1,J)
      IF(IX1.EQ.0.AND.IY1.EQ.0) THEN
      IT(I,J)=0
      GOTO 200
      ENDIF 
      IF(IY1.EQ.0.AND.IX2.EQ.0) THEN
      IT(I,J)=0
      GOTO 200
      ENDIF
      IF(IX2.EQ.0.AND.IY2.EQ.0) THEN
      IT(I,J)=0
      GOTO 200 
      ENDIF
      IF(IY2.EQ.0.AND.IX1.EQ.0) IT(I,J)=0
  200 CONTINUE
C********   AIDX & AIDY *********
      DO I=1,IJS
      AIDX(I,1)=1.
      AIDY(I,1)=1.
      IF(IDX(I,1).EQ.0) AIDX(I,1)=0.
      IF(IDY(I,1).EQ.0) AIDY(I,1)=0.
      ENDDO
C*********************************************************
c      open(11,file='bot-uv',form='unformatted')
c      write(11) hu,hv,ht
c      close(11)
C
 2222 CONTINUE
      RETURN
      END


C     This routine calculates 3D tides over Kashevarov Bank.

      INCLUDE 'comblk'

      DATA Z/2.5,5.,10.,15.,20.,25.,30.,40.,50.,75.,80.,90.,100.,110.,

      data HST/0.05,0.2,0.5,0.95,1.6,3.5/
