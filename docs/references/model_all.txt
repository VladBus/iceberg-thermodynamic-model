      SUBROUTINE ADV2D(IS,JS,IS1,JS1,DT,DX,C2,KT1,IT,
     *IDX,IDY,U,V,AN3,APX2,APY2,CD2)
C
      REAL U(IS1,JS1),V(IS1,JS1),AN3(IS1,JS1),
     *CD2(IS1,JS1),APX2(IS1,JS1),APY2(IS1,JS1)
      INTEGER KT1(IS1,JS1),IDX(IS1,JS1),IDY(IS1,JS1),IT(IS1,JS1)
      IJS1=IS1*JS1
C
      DO I=1,IJS1
      APX2(I,1)=0.
      APY2(I,1)=0.
      ENDDO
C*************************************************
      DO 11 I=1,IS
      DO 11 J=1,JS
C--------------------------------------------------
      IF(KT1(I,J).EQ.0.OR.IT(I,J).NE.0) GOTO 11
      I1=I+1
      J1=J+1
      UP=50.*(U(I,J1)+U(I1,J1))
      UM=50.*(U(I,J)+U(I1,J))
      UU=0.5*(UP+UM)
      VP=50.*(V(I1,J)+V(I1,J1))
      VM=50.*(V(I,J)+V(I,J1))
      VV=0.5*(VP+VM)
      IY1=IDY(I1,J)
      IX1=IDX(I,J1)
C--------------------------------------------------
      CC=AN3(I,J)
      IF(IY1.EQ.0) THEN
      CI1=0.
      I3=I1
      ELSE
      CI1=AN3(I1,J)
      I3=I+2
      ENDIF
      IF(IDY(I,J).EQ.0) THEN
      CI2=0.
      ELSE
      CI2=AN3(I-1,J)
      ENDIF
      IF(IX1.EQ.0) THEN
      CJ1=0.
      J3=J1
      ELSE
      CJ1=AN3(I,J1)
      J3=J+2
      ENDIF
      IF(IDX(I,J).EQ.0) THEN
      CJ2=0.
      ELSE
      CJ2=AN3(I,J-1)
      ENDIF
C-----------------------------------------------------
      CDX=0.
      IF(UU.GT.0.) THEN
      CDX=(CC-CJ2)*UU
      ELSE
      CDX=(CJ1-CC)*UU
      ENDIF
      IF(CDX.NE.0.) GOTO 201
      IF(UM.GT.0.)  CDX=(CC-CJ2)*UM
      IF(UP.LT.0.)  CDX=(CJ1-CC)*UP
  201 CONTINUE
      CDX=CDX+CC*(UP-UM)*0.5
C-----------------------------------------------------
      CDY=0.
      IF(VV.GT.0.) THEN
      CDY=(CC-CI1)*VV
      ELSE
      CDY=(CI2-CC)*VV
      ENDIF
      IF(CDY.NE.0.) GOTO 202
      IF(VP.GT.0.)  CDY=(CC-CI1)*VP
      IF(VM.LT.0.)  CDY=(CI2-CC)*VM
  202 CONTINUE
      CDY=CDY+CC*(VM-VP)*0.5
C-------------------------------------------------------
      CD2(I,J)=CC-DT*(CDX+CDY)/DX
C----------------------------------------------------
      IF(IX1.NE.0) THEN
      IF(UP.LT.0.) THEN
      FLXP=CJ1*UP
      ELSE
      FLXP=CC*UP
      ENDIF
      IF(J.NE.JS) THEN
      J3=J+2
      IF(IDX(I,J3).NE.0) THEN
      CJ3=AN3(I,J3)
      ELSE
      CJ3=0.
      ENDIF
      ENDIF
      APX2(I,J1)=(UP*(7.*(CC+CJ1)-(CJ2+CJ3))/12.-FLXP)*C2
      ENDIF
      IF(IY1.NE.0) THEN
         IF(VP.LT.0.) THEN
         FLYP=CC*VP
         ELSE
         FLYP=CI1*VP
         ENDIF
         IF(I.NE.IS) THEN
         I3=I+2
         IF(IDY(I3,J).NE.0) THEN
         CI3=AN3(I3,J)
         ELSE
         CI3=0.
         ENDIF
         ENDIF
         APY2(I1,J)=(VP*(7.*(CC+CI1)-(CI2+CI3))/12.-FLYP)*C2
      ENDIF
   11 CONTINUE
C********************************************************
      DO 91 I=1,IS
      DO 91 J=1,JS
      IF(IDX(I,J).EQ.0) GOTO 91
      J1=J+1
      J2=J-1
      IX1=IDX(I,J1)
      IX2=IDX(I,J2)
      IF(IX1.EQ.0) THEN
      A1=0.
      ELSE
      A1=CD2(I,J1)-CD2(I,J)
      ENDIF
      IF(IX2.EQ.0) THEN
      B1=0.
      ELSE
      B1=CD2(I,J2)-CD2(I,J-2)
      ENDIF
      A=APX2(I,J)
      B=SIGN(1.,A)
      APX2(I,J)=B*AMAX1(0.,AMIN1(ABS(A),B*A1,B*B1))
   91 CONTINUE
C**********************************************************
      DO 191 J=1,JS
      J1=J+1
      DO 191 I=1,IS
      IF(KT1(I,J).EQ.0.OR.IT(I,J).NE.0) GOTO 191
      CD2(I,J)=CD2(I,J)-(APX2(I,J1)-APX2(I,J))
  191 CONTINUE
C******* y-COORDINATE
      DO 391 I=1,IS
      DO 391 J=1,JS
      IF(IDY(I,J).EQ.0) GOTO 391
      I1=I+1
      I2=I-1
      IY1=IDY(I1,J)
      IY2=IDY(I2,J)
      IF(IY1.EQ.0) THEN
      A1=0.
      ELSE
      A1=CD2(I,J)-CD2(I1,J)
      ENDIF
      IF(IY2.EQ.0) THEN
      B1=0.
      ELSE
      B1=CD2(I-2,J)-CD2(I2,J)
      ENDIF
      A=APY2(I,J)
      B=SIGN(1.,A)
      APY2(I,J)=B*AMAX1(0.,AMIN1(ABS(A),B*A1,B*B1))
  391 CONTINUE
C**********************************************************
      DO 392 I=1,IS
      I1=I+1
      DO 392 J=1,JS
      IF(KT1(I,J).EQ.0.OR.IT(I,J).NE.0) GOTO 392
      AN3(I,J)=CD2(I,J)-(APY2(I,J)-APY2(I1,J))
  392 CONTINUE
C*******************************************************
      DO 400 I=1,IS
      DO 400 J=1,JS
      KI=IT(I,J)
      IF(KI.EQ.0.OR.KI.EQ.9) GOTO 400
      IF(KI.EQ.1) THEN
      AN3(I,J)=0.5*(AN3(I,J+1)+AN3(I+1,J))
      GOTO 400
      ENDIF
      IF(KI.EQ.2) THEN
      AN3(I,J)=0.5*(AN3(I,J-1)+AN3(I+1,J))
      GOTO 400
      ENDIF
      IF(KI.EQ.3) THEN
      AN3(I,J)=0.5*(AN3(I,J-1)+AN3(I-1,J))
      GOTO 400
      ENDIF
      IF(KI.EQ.4) THEN
      AN3(I,J)=0.5*(AN3(I,J+1)+AN3(I-1,J))
      GOTO 400
      ENDIF
  400 CONTINUE
C******************************************************  
      RETURN
      END


      SUBROUTINE ADVS(IS,JS,KS,IS1,JS1,KS1,DT,DX,C2,U2,V2,
     *HT,Z,KT1,IDX,IDY,DZ,DZ1,W,S1,S2,APX,APY,APZ,ADX,ADY,CD,TT)
C
      DIMENSION S1(IS1,JS1,KS),
     *HT(IS1,JS1),CD(IS1,JS1,KS),W(IS1,JS1,KS1),DZ1(KS),DZ(KS1),
     *Z(KS),APX(IS1,JS1,KS),APY(IS1,JS1,KS),APZ(IS1,JS1,KS1),
     *S2(IS1,JS1,KS),TT(KS),U2(IS1,JS1,KS),V2(IS1,JS1,KS),
     *ADX(IS1,JS1,KS1),ADY(IS1,JS1,KS1)
      INTEGER KT1(IS1,JS1),IDX(IS1,JS1),IDY(IS1,JS1)
C**************************************************
      IJKS=IS1*JS1*KS
                                    !DO I=1,IS1*JS1*KS1
      APZ=0.                        !APZ(I,1,1)=0.
                                    !ENDDO
                                    !DO I=1,IJKS
      APX=0.                        !APX(I,1,1)=0.
      APY=0.                        !APY(I,1,1)=0.
                                    !ENDDO
C
      DO 11 J=1,JS
      J1=J+1
      J2=J-1
      DO 11 I=1,IS
C--------------------------------------------------
      KI=KT1(I,J)
      IF(KI.EQ.0) GOTO 11
      I1=I+1
      I2=I-1
      DO 10 K=1,KI
C--------------------------------------------------
      CC= S2(I,J,K)
      CI1=S2(I1,J,K)
      CI2=S2(I2,J,K)
      CJ1=S2(I,J1,K)
      CJ2=S2(I,J2,K)
      UP=U2(I,J1,K)
      UU=0.5*(UP+U2(I,J,K))
      VP=V2(I1,J,K)
      VV=0.5*(VP+V2(I,J,K))
C-----------------------------------------------------
      A=ABS(UU)
      CDX=0.5*(UU+A)*ADX(I,J,K)*(CC-CJ2)
     *+0.5*(UU-A)*ADX(I,J1,K)*(CJ1-CC)
      A=ABS(VV)
      CDY=0.5*(VV+A)*ADY(I1,J,K)*(CC-CI1)
     *+0.5*(VV-A)*ADY(I,J,K)*(CI2-CC)
      TT(K)=S1(I,J,K)-C2*(CDX+CDY)
C------------------------------------------------------
      A=ABS(UP)
      FLXP=0.5*(UP+A)*CC+0.5*(UP-A)*CJ1
      APX(I,J1,K)=ADX(I,J1,K)*(UP*0.5*(CC+CJ1)-FLXP)*C2
      A=ABS(VP)       
      FLYP=0.5*(VP+A)*CC+0.5*(VP-A)*CI1
      APY(I+1,J,K)=ADY(I1,J,K)*(VP*0.5*(CC+CI1)-FLYP)*C2
C-------------------------------------------------------
   10 CONTINUE
C-------------------------------------------------------
      CK2=S2(I,J,1)
      CC=CK2
      WW1=W(I,J,1)
      DO K=1,KI-1   
      K1=K+1
      WW2=W(I,J,K1)
      WW=0.5*(WW1+WW2)
      A=ABS(WW)     
      CK1=S2(I,J,K1)
      CD(I,J,K)=TT(K)-DT*
     *(0.5*(WW+A)*(CC-CK2)/DZ(K)+0.5*(WW-A)*(CK1-CC)/DZ(K1))
      CK2=CC
      CC=CK1
      WW1=WW2 
      ENDDO  
      CD(I,J,KI)=TT(KI) 
      CC=S2(I,J,1)
      DO K=2,KI
      CK1=S2(I,J,K)
      WW=W(I,J,K)
      A=ABS(WW)
      APZ(I,J,K)=0.5*WW*(CC+CK1)-0.5*(WW+A)*CC+0.5*(WW-A)*CK1
      CC=CK1
      ENDDO
   11 CONTINUE
C********************************************************
      DO 91 J=1,JS
      DO 91 I=1,IS
      KI=IDX(I,J)
      IF(KI.EQ.0)GO TO 91
      J1=J+1
      J2=J-1
      DO K=1,KI
      A1=ADX(I,J1,K)*(CD(I,J1,K)-CD(I,J,K))
      B1=ADX(I,J2,K)*(CD(I,J2,K)-CD(I,J-2,K))
      A=APX(I,J,K)
      B=SIGN(1.,A)
      APX(I,J,K)=B*AMAX1(0.,AMIN1(ABS(A),B*A1,B*B1))
      ENDDO
   91 CONTINUE
C**********************************************************
      DO 191 J=1,JS
      J1=J+1
      DO 191 I=1,IS
      KI=KT1(I,J)
      IF(KI.EQ.0)GO TO 191
      DO K=1,KI
      CD(I,J,K)=CD(I,J,K)-(APX(I,J1,K)-APX(I,J,K))
      ENDDO
  191 CONTINUE
C******* y-COORDINATE
      DO 391 I=1,IS
      I1=I+1
      I2=I-1
      DO 391 J=1,JS
      KI=IDY(I,J)
      IF(KI.EQ.0)GO TO 391
      DO K=1,KI
      A1=ADY(I1,J,K)*(CD(I,J,K)-CD(I1,J,K))
      B1=ADY(I2,J,K)*(CD(I-2,J,K)-CD(I2,J,K))
      A=APY(I,J,K)
      B=SIGN(1.,A)
      APY(I,J,K)=B*AMAX1(0.,AMIN1(ABS(A),B*A1,B*B1))
      ENDDO
  391 CONTINUE
C**********************************************************
      DO 392 I=1,IS 
      I1=I+1
      DO 392 J=1,JS
      KI=KT1(I,J)
      IF(KI.EQ.0)GO TO 392
      DO K=1,KI
      CD(I,J,K)=CD(I,J,K)-(APY(I,J,K)-APY(I1,J,K))
      ENDDO
  392 CONTINUE
C*******************************************************
      DO 491 I=1,IS
      DO 491 J=1,JS
      KI=KT1(I,J)
      IF(KI.EQ.0)GO TO 491
      B1=(CD(I,J,3)-CD(I,J,2))*DZ1(2)/DT
      A=APZ(I,J,2)
      B=SIGN(1.,A)
      APZ(I,J,2)=B*AMAX1(0.,AMIN1(ABS(A),B*B1))
      DO K=3,KI-1
      A1=(CD(I,J,K-1)-CD(I,J,K-2))*DZ1(K-1)/DT
      B1=(CD(I,J,K+1)-CD(I,J,K))*DZ1(K)/DT
      A=APZ(I,J,K)
      B=SIGN(1.,A)
      APZ(I,J,K)=B*AMAX1(0.,AMIN1(ABS(A),B*A1,B*B1))
      ENDDO
      A1=(CD(I,J,KI-1)-CD(I,J,KI-2))*DZ1(KI-1)/DT
      A=APZ(I,J,KI)
      B=SIGN(1.,A)
      APZ(I,J,KI)=B*AMAX1(0.,AMIN1(ABS(A),B*A1))
  491 CONTINUE
C**********************************************************
      DO 591 I=1,IS
      DO 591 J=1,JS
      KI=KT1(I,J)
      IF(KI.EQ.0)GO TO 591
      DO K=2,KI-1
      CD(I,J,K)=CD(I,J,K)-(APZ(I,J,K+1)-APZ(I,J,K))*DT/DZ1(K)
      ENDDO
  591 CONTINUE
C
                                          !DO I=1,IJKS
      S2=CD                               !S2(I,1,1)=CD(I,1,1)
                                          !ENDDO
C*******************************************************
      RETURN
      END


      SUBROUTINE ADVSH(IS,JS,IS1,JS1,IS2,JS2,DT1,DX,C10,
     *HU,HV,KUSH,KVSH,UP2,VP2,APX2,APY2,CD2)
C
      REAL UP2(IS1,JS1),VP2(IS1,JS1),HU(IS1,JS1),HV(IS1,JS1),
     *CD2(IS1,JS1),APX2(IS1,JS1),APY2(IS1,JS1)
      INTEGER KUSH(IS1,JS1),KVSH(IS1,JS1)
      IJS1=IS1*JS1
C
                                 !DO I=1,IJS1
      CD2=0.                     !CD2(I,1)=0.
      APX2=0.                    !APX2(I,1)=0.
      APY2=0.                    !APY2(I,1)=0.
                                 !ENDDO
C*************************************************
      DO 11 I=2,IS2
      I1=I+1
      I2=I-1
      DO 11 J=3,JS2
C--------------------------------------------------
      HHT=HU(I,J)
      IF(HHT.EQ.8888.) GOTO 11
      J1=J+1
      J2=J-1
      UIJ=UP2(I,J)
      UU1=UP2(I,J1)
      UU2=UP2(I,J2)
      UU3=UP2(I2,J)
      UU4=UP2(I1,J)
      IF(KUSH(I,J).NE.0) THEN
      V11=VP2(I,J)/HV(I,J)+VP2(I,J2)/HV(I,J2)
      V12=VP2(I1,J)/HV(I1,J)+VP2(I1,J2)/HV(I1,J2)
      A=UIJ/HHT
      U11=UU1/HU(I,J1)+A
      U12=A+UU2/HU(I,J2)
      CDX=U11*(UIJ+UU1)+ABS(U11)*(UIJ-UU1)-
     *U12*(UIJ+UU2)-ABS(U12)*(UU2-UIJ)
      CDY=V11*(UIJ+UU3)+ABS(V11)*(UIJ-UU3)-
     *V12*(UIJ+UU4)-ABS(V12)*(UU4-UIJ)
      CD2(I,J)=UIJ-C10*0.5*(CDX+CDY*0.)
      ENDIF
C----------------------------------------------------
      IF(J.NE.JS) THEN
      IF(UJP.GE.0.) THEN
      FLXP=UJP*UU
      ELSE
      FLXP=UJP*UJ1
      ENDIF
      APX2(I,J1)=(UJP*UJP-FLXP)*A
      ENDIF
         IF(I.NE.IS) THEN
         IF(VIP.GE.0.) THEN
         FLYP=VIP*UI1
         ELSE
         FLYP=VIP*UU
         ENDIF
         APY2(I1,J)=(UIP*VIP-FLYP)*A
         ENDIF
   11 CONTINUE
C********************************************************
      DO 21 I=2,IS2
      DO 21 J=3,JS
      J2=J-1
      A2=HU(I,J)
      B2=HU(I,J2)
      IF(A2.EQ.8888..OR.B2.EQ.8888.) THEN
      APX2(I,J)=0.
      ELSE
      A1=CD2(I,J+1)-CD2(I,J)
      B1=CD2(I,J2)-CD2(I,J-2)
      A=APX2(I,J)
      B=SIGN(1.,A)
      APX2(I,J)=B*AMAX1(0.,AMIN1(ABS(A),B*A1,B*B1))
      ENDIF
   21 CONTINUE
C**********************************************************
      DO 31 J=3,JS2
      J1=J+1
      DO 31 I=2,IS2
      IF(KUSH(I,J).EQ.0) GOTO 31
      CD2(I,J)=CD2(I,J)-(APX2(I,J1)-APX2(I,J))
   31 CONTINUE
C******* y-COORDINATE
      DO 41 I=3,IS2
      I2=I-1
      DO 41 J=3,JS2
      A2=HU(I,J)
      B2=HU(I2,J)
      IF(A2.EQ.8888..OR.B2.EQ.8888.) THEN
      APY2(I,J)=0.
      ELSE
      A1=CD2(I,J)-CD2(I1,J)
      B1=CD2(I-2,J)-CD2(I2,J)
      A=APY2(I,J)
      B=SIGN(1.,A)
      APY2(I,J)=B*AMAX1(0.,AMIN1(ABS(A),B*A1,B*B1))
      ENDIF
   41 CONTINUE
C**********************************************************
      DO 51 I=2,IS2
      I1=I+1
      DO 51 J=3,JS2
      IF(KUSH(I,J).EQ.0) GOTO 51
      UP2(I,J)=CD2(I,J)-(APY2(I,J)-APY2(I1,J))
   51 CONTINUE
C******************************************************  
                               !DO I=1,IJS1
      CD2=0.                   !CD2(I,1)=0.
      APX2=0.                  !APX2(I,1)=0.
      APY2=0.                  !APY2(I,1)=0.
                               !ENDDO
C*************************************************
      DO 111 I=3,IS2
      I1=I+1
      I2=I-1
      DO 111 J=2,JS2
C--------------------------------------------------
      HHT=HV(I,J)
      IF(HHT.EQ.8888.) GOTO 111
      J1=J+1
      J2=J-1
      VIJ=VP2(I,J)
      VV1=VP2(I2,J)
      VV2=VP2(I1,J)
      VV3=VP2(I,J1)
      VV4=VP2(I,J2)
      IF(KVSH(I,J).NE.0) THEN
      U11=UP2(I2,J1)/HU(I2,J1)+UP2(I,J1)/HU(I,J1)
      U12=UP2(I,J)/HU(I,J)+UP2(I2,J)/HU(I2,J)
      A=VIJ/HHT
      V11=VV1/HV(I2,J)+A
      V12=A+VV2/HV(I1,J)
      CDX=U11*(VIJ+VV3)+ABS(U11)*(VIJ-VV3)-
     *U12*(VIJ+VV4)-ABS(U12)*(VV4-VIJ)
      CDY=V11*(VIJ+VV1)+ABS(V11)*(VIJ-VV1)-
     *V12*(VIJ+VV2)-ABS(V12)*(VV2-VIJ)
      CD2(I,J)=VIJ-C10*0.5*(CDX+CDY)
      ENDIF
C----------------------------------------------------
      IF(J.NE.JS) THEN
      IF(UJP.GE.0.) THEN
      FLXP=UJP*VV
      ELSE
      FLXP=UJP*VJ1
      ENDIF
      APX2(I,J1)=(UJP*VJP-FLXP)*A
      ENDIF
         IF(I.NE.IS) THEN
         IF(VIP.GE.0.) THEN
         FLYP=VIP*VV
         ELSE
         FLYP=VIP*VI2
         ENDIF
         APY2(I1,J)=(VIP*VIP-FLYP)*A
         ENDIF
  111 CONTINUE
C********************************************************
      DO 121 I=3,IS2
      DO 121 J=3,JS2
      J2=J-1
      A2=HV(I,J)
      B2=HV(I,J2)
      IF(A2.EQ.8888..OR.B2.EQ.8888.) THEN
      APX2(I,J)=0.
      ELSE
      A1=CD2(I,J+1)-CD2(I,J)
      B1=CD2(I,J2)-CD2(I,J-2)
      A=APX2(I,J)
      B=SIGN(1.,A)
      APX2(I,J)=B*AMAX1(0.,AMIN1(ABS(A),B*A1,B*B1))
      ENDIF
  121 CONTINUE
C**********************************************************
      DO 131 J=2,JS2
      J1=J+1
      DO 131 I=3,IS2
      IF(KVSH(I,J).EQ.0) GOTO 131
      CD2(I,J)=CD2(I,J)-(APX2(I,J1)-APX2(I,J))
  131 CONTINUE
C******* y-COORDINATE
      DO 141 I=3,IS
      I2=I-1
      DO 141 J=2,JS2
      A2=HV(I,J)
      B2=HV(I2,J)
      IF(A2.EQ.8888..OR.B2.EQ.8888.) THEN
      APY2(I,J)=0.
      ELSE
      A1=CD2(I,J)-CD2(I1,J)
      B1=CD2(I-2,J)-CD2(I2,J)
      A=APY2(I,J)
      B=SIGN(1.,A)
      APY2(I,J)=B*AMAX1(0.,AMIN1(ABS(A),B*A1,B*B1))
      ENDIF
  141 CONTINUE
C**********************************************************
      DO 151 I=3,IS2
      I1=I+1
      DO 151 J=2,JS2
      IF(KVSH(I,J).EQ.0) GOTO 151
      VP2(I,J)=CD2(I,J)-(APY2(I,J)-APY2(I1,J))
  151 CONTINUE
C******************************************************  
      RETURN
      END


      SUBROUTINE ADVT(IS,JS,KS,IS1,JS1,KS1,DT,DX,C2,U2,V2,
     *HT,Z,KT1,IDX,IDY,DZ,DZ1,W,T1,T2,APX,APY,APZ,ADX,ADY,CD,TT)
C
      DIMENSION T1(IS1,JS1,KS),
     *HT(IS1,JS1),CD(IS1,JS1,KS),W(IS1,JS1,KS1),DZ1(KS),DZ(KS1),
     *Z(KS),APX(IS1,JS1,KS),APY(IS1,JS1,KS),APZ(IS1,JS1,KS1),
     *T2(IS1,JS1,KS),TT(KS),U2(IS1,JS1,KS),V2(IS1,JS1,KS),
     *ADX(IS1,JS1,KS1),ADY(IS1,JS1,KS1)
      INTEGER KT1(IS1,JS1),IDX(IS1,JS1),IDY(IS1,JS1)

      IJKS=IS1*JS1*KS
                                !DO I=1,IS1*JS1*KS1
      APZ=0.                    !APZ(I,1,1)=0.

                                !DO I=1,IJKS
      APX=0.                    !APX(I,1,1)=0.
      APY=0.                    !APY(I,1,1)=0.

      DO 11 J=1,JS
      J1=J+1
      J2=J-1
      DO 11 I=1,IS
C--------------------------------------------------
      KI=KT1(I,J)
      IF(KI.EQ.0) GOTO 11
      I1=I+1
      I2=I-1
      DO 10 K=1,KI
C--------------------------------------------------
      CC= T2(I,J,K)
      CI1=T2(I1,J,K)
      CI2=T2(I2,J,K)
      CJ1=T2(I,J1,K)
      CJ2=T2(I,J2,K)
      UP=U2(I,J1,K)
      UU=0.5*(UP+U2(I,J,K))
      VP=V2(I1,J,K)
      VV=0.5*(VP+V2(I,J,K))
C-----------------------------------------------------
      A=ABS(UU)
      CDX=0.5*(UU+A)*ADX(I,J,K)*(CC-CJ2)
     *+0.5*(UU-A)*ADX(I,J1,K)*(CJ1-CC)
      A=ABS(VV)
      CDY=0.5*(VV+A)*ADY(I1,J,K)*(CC-CI1)
     *+0.5*(VV-A)*ADY(I,J,K)*(CI2-CC)
      TT(K)=T1(I,J,K)-C2*(CDX+CDY)
C------------------------------------------------------
      A=ABS(UP)
      FLXP=0.5*(UP+A)*CC+0.5*(UP-A)*CJ1
      APX(I,J1,K)=ADX(I,J1,K)*(UP*0.5*(CC+CJ1)-FLXP)*C2
      A=ABS(VP)       
      FLYP=0.5*(VP+A)*CC+0.5*(VP-A)*CI1
      APY(I+1,J,K)=ADY(I1,J,K)*(VP*0.5*(CC+CI1)-FLYP)*C2
C-------------------------------------------------------
   10 CONTINUE
C-------------------------------------------------------
      CK2=T2(I,J,1)
      CC=CK2
      WW1=W(I,J,1)

      WW2=W(I,J,K1)
      WW=0.5*(WW1+WW2)
      A=ABS(WW)     
      CK1=T2(I,J,K1)
      CD(I,J,K)=TT(K)-DT*
     *(0.5*(WW+A)*(CC-CK2)/DZ(K)+0.5*(WW-A)*(CK1-CC)/DZ(K1))
      CK2=CC
      CC=CK1
      WW1=WW2 
      ENDDO  
      CD(I,J,KI)=TT(KI) 
      CC=T2(I,J,1)
      DO K=2,KI
      CK1=T2(I,J,K)
      WW=W(I,J,K)
      A=ABS(WW)
      APZ(I,J,K)=0.5*WW*(CC+CK1)-0.5*(WW+A)*CC+0.5*(WW-A)*CK1
      CC=CK1
      ENDDO
   11 CONTINUE
C********************************************************
      DO 91 J=1,JS
      DO 91 I=1,IS
      KI=IDX(I,J)
      IF(KI.EQ.0) GOTO 91
      J1=J+1
      J2=J-1
      DO K=1,KI
      A1=ADX(I,J1,K)*(CD(I,J1,K)-CD(I,J,K))
      B1=ADX(I,J2,K)*(CD(I,J2,K)-CD(I,J-2,K))
      A=APX(I,J,K)
      B=SIGN(1.,A)
      APX(I,J,K)=B*AMAX1(0.,AMIN1(ABS(A),B*A1,B*B1))
      ENDDO
   91 CONTINUE
C**********************************************************
      DO 191 J=1,JS
      J1=J+1
      DO 191 I=1,IS
      KI=KT1(I,J)
      IF(KI.EQ.0) GOTO 191
      DO K=1,KI
      CD(I,J,K)=CD(I,J,K)-(APX(I,J1,K)-APX(I,J,K))
      ENDDO
  191 CONTINUE
C******* y-COORDINATE
      DO 391 I=1,IS
      I1=I+1
      I2=I-1
      DO 391 J=1,JS
      KI=IDY(I,J)
      IF(KI.EQ.0) GOTO 391
      DO K=1,KI
      A1=ADY(I1,J,K)*(CD(I,J,K)-CD(I1,J,K))
      B1=ADY(I2,J,K)*(CD(I-2,J,K)-CD(I2,J,K))
      A=APY(I,J,K)
      B=SIGN(1.,A)
      APY(I,J,K)=B*AMAX1(0.,AMIN1(ABS(A),B*A1,B*B1))
      ENDDO
  391 CONTINUE
C**********************************************************
      DO 392 I=1,IS 
      I1=I+1
      DO 392 J=1,JS
      KI=KT1(I,J)
      IF(KI.EQ.0) GOTO 392
      DO K=1,KI
      CD(I,J,K)=CD(I,J,K)-(APY(I,J,K)-APY(I1,J,K))
      ENDDO
  392 CONTINUE
C*******************************************************
      DO 491 I=1,IS
      DO 491 J=1,JS
      KI=KT1(I,J)
      IF(KI.EQ.0) GOTO 491
      B1=(CD(I,J,3)-CD(I,J,2))*DZ1(2)/DT
      A=APZ(I,J,2)
      B=SIGN(1.,A)
      APZ(I,J,2)=B*AMAX1(0.,AMIN1(ABS(A),B*B1))
      DO K=3,KI-1
      A1=(CD(I,J,K-1)-CD(I,J,K-2))*DZ1(K-1)/DT
      B1=(CD(I,J,K+1)-CD(I,J,K))*DZ1(K)/DT
      A=APZ(I,J,K)
      B=SIGN(1.,A)
      APZ(I,J,K)=B*AMAX1(0.,AMIN1(ABS(A),B*A1,B*B1))
      ENDDO
      A1=(CD(I,J,KI-1)-CD(I,J,KI-2))*DZ1(KI-1)/DT
      A=APZ(I,J,KI)
      B=SIGN(1.,A)
      APZ(I,J,KI)=B*AMAX1(0.,AMIN1(ABS(A),B*A1))
  491 CONTINUE
C**********************************************************
      DO 591 I=1,IS
      DO 591 J=1,JS
      KI=KT1(I,J)
      IF(KI.EQ.0) GOTO 591
      DO K=2,KI-1
      CD(I,J,K)=CD(I,J,K)-(APZ(I,J,K+1)-APZ(I,J,K))*DT/DZ1(K)
      ENDDO
  591 CONTINUE
C
                                !DO I=1,IJKS
      T2=CD                     !T2(I,1,1)=CD(I,1,1)
                                !ENDDO
C*******************************************************
      RETURN
      END

      DZZ=DZ(1)

      DO 201 K=1,KI

C      B1=RI2J2+RI2J-RIJ2-RIJ

      DO 210 I=2,IS

      YYY=0.25*(YM2(I2,J2)+YM2(I,J2)+YM2(I2,J)+YM2(I,J))

      DO 211 K=1,KI

      A1=0.25*(ANS(I,J)+ANS(I,J2)+ANS(I2,J)+ANS(I2,J2))

      DO 215 K=2,KI2

      AA=DT/DZ1(K)

      UCA(K)=B/AA

      AA=A1-VCA(K2)*A
      VCA(K)=B/AA
  215 VNU(K)=(A*VNU(K2)-V2(I,J,K))/AA
C    i n v e r s e    s t e p
  214 DZZ=HHT-0.5*(Z(KI)+Z(KI2))
      DZZZ=HHT-Z(KI)+50.
      B=DT/DZZ/DZ(KI)*RR(KI2)

      BB=B/A
      U2(I,J,KI)=(U2(I,J,KI)/A+BB*UNU(KI2))/(1.-BB*UCA(KI2))
      V2(I,J,KI)=(V2(I,J,KI)/A+BB*VNU(KI2))/(1.-BB*VCA(KI2))
      DO 213 K=1,KI2
      KK=KI-K
      K1=KK+1
      U2(I,J,KK)=UCA(KK)*U2(I,J,K1)+UNU(KK)
  213 V2(I,J,KK)=VCA(KK)*V2(I,J,K1)+VNU(KK)
      GOTO 210
  212 U2(I,J,KI)=UCA(KI)*U2(I,J,KI1)+UNU(KI)
      V2(I,J,KI)=VCA(KI)*V2(I,J,KI1)+VNU(KI)
  210 CONTINUE
      write(2,*)' vertical viscosity'
      write(*,*)' vertical viscosity'
C========================================================================
C    calculation of the vertically mean currents
C========================================================================
      CALL shal(IS,JS,KS,IS1,JS1,IS2,JS2,LI,LJ,DT1,DX,C1,C3,C9,C10,MM3,
     *HT,HU,HV,KT1,KUSH,KVSH,Y2,UP1,UP2,VP1,VP2,YM2,

      write(*,*)' shallow'
      write(2,*)' shallow'
C========================================================================
C      calculation of current velocities  U and  V
C========================================================================
      DO 280 I=2,IS
      I2=I-1
      DO 280 J=2,JS
      KI=KK1(I,J)

      J2=J-1
      SUM=0.
      SUM1=0.
      HHT=map1(I,J)
      DO K=1,KI
      K1=K+1
      IF(K.EQ.KI) THEN
      IF(KI.NE.1) THEN
      DZZ=HHT-0.5*(Z(KI)+Z(KI-1))
      ELSE
      DZZ=HHT
      ENDIF
      ELSE
      DZZ=DZ1(K)
      ENDIF
      SUM=U2(I,J,K)*DZZ+SUM

      SUM=(-SUM+0.5*(UP2(I,J)+UP2(I2,J)))/HHT
      SUM1=(-SUM1+0.5*(VP2(I,J)+VP2(I,J2)))/HHT
      DO K=1,KI
      U2(I,J,K)=U2(I,J,K)+SUM
      V2(I,J,K)=V2(I,J,K)+SUM1

  280 CONTINUE
C---------------------- BOUNDARY CONDISIONS ---------------
                                !DO I=1,IS1

      U2(:,1,:)=U2(:,2,:)       !U2(I,1,K)=U2(I,2,K)

                                !ENDDO

      U2(1,:,:)=U2(2,:,:)       !U2(1,J,K)=U2(2,J,K)
      V2(1,:,:)=V2(2,:,:)       !V2(1,J,K)=V2(2,J,K)
                                !ENDDO
                                !ENDDO
C*******************************************************************
      IKKK=IKKK+1
      YYV(IKKK)=Y2(56,53)
      UF1(IKKK)=U(56,53)
      VF1(IKKK)=V(56,53)
      UF2(IKKK)=U(57,53)
      VF2(IKKK)=V(57,53)

      VVF(IKKK)=V2(57,53,1)
      WIU(IKKK)=WINDX(57,53)
      WIV(IKKK)=WINDY(57,53)
                                  !DO I=1,NGR1

      SUBROUTINE DEFORM(IS,JS,IS1,JS1,IS3,JS3,IJS1,DX,DT1,KT1,
     *EPR,U,V,CD2,KL)
      REAL EPR(IS3,JS3),U(IS1,JS1),V(IS1,JS1),CD2(IS1,JS1)
      INTEGER KT1(IS1,JS1)
C
      DO 1 J=1,JS
      J1=J+1
      A1=0.
      DO 1 I=1,IS
      IF(KT1(I,J).EQ.0) GOTO 1
      I1=I+1
      V1=V(I,J)
      V2=V(I,J1)
      V3=V(I1,J)
      V4=V(I1,J1)
      IF(KL-2) 2,3,4
    2 CD2(I,J)=50.*(U(I,J1)+U(I1,J1)-U(I,J)-U(I1,J))/DX
      GOTO 5
    3 CD2(I,J)=50.*(V1+V2-V3-V4)/DX
      GOTO 5
    4 CD2(I,J)=25.*(U(I,J)+U(I,J1)-U(I1,J)-U(I1,J1)+V2+V4-V1-V3)/DX
    5 B=0.25*(V1+V2+V3+V4)
      B1=0.25*(U(I,J)+U(I,J1)+U(I1,J1)+U(I1,J))
      A=EPR(I1,J1)
      IF(B1.GE.0.) THEN
      A1=B1*(A-EPR(I1,J))
      ELSE
      A1=B1*(EPR(I1,J+2)-A)
      ENDIF
      IF(B.GE.0.) THEN
      A2=B*(A-EPR(I+2,J))
      ELSE
      A2=B*(EPR(I,J1)-A)
      ENDIF
      EPR(I1,J1)=A-DT1*(100./DX*(A1+A2)-CD2(I,J))
    1 CONTINUE
C
      RETURN
      END


      subroutine HEAT(III,IS1,JS1,KS,KS1,NGR,NGR1,NDAY,DT,KT1,AN1,ANS,
     *WICE1,WICES,HICE,HICES,HSNOW,TPAR,SPAR,ANP,DANP,HSNP,HICP,SICST,
     *T1,S1,SKT,DZ,DZ1,FI,PATM,TATM,WIND,CLOUD,HUMID,ALSN,SFAL,LLL,fw)
C
      REAL DZ(KS1),DZ1(KS),FI(IS1,JS1),T1(IS1,JS1,KS),S1(IS1,JS1,KS),
     *SKT(IS1,JS1),WICE1(IS1,JS1,NGR),AN1(IS1,JS1,NGR1),
     *HICE(IS1,JS1,NGR),WICES(IS1,JS1),HICES(IS1,JS1),ANS(IS1,JS1),
     *HSNOW(IS1,JS1,NGR),TPAR(NGR1),SPAR(NGR1),SICST(NGR),ALSN(12),
     *SFAL(12),ANP(NGR),DANP(NGR),HSNP(NGR),HICP(NGR)
      REAL CLOUD(IS1,JS1),HUMID(IS1,JS1),PATM(IS1,JS1),
     *TATM(IS1,JS1),WIND(IS1,JS1)
      INTEGER KT1(IS1,IS1)
C
      DZZ=DZ1(1)/100.
      DTZ=DT/(DZZ*4.19E6)
      JS=JS1-1
      IS=IS1-1
      DO 1 I=1,IS
      DO 1 J=1,JS
      IF(KT1(I,J).EQ.0.OR.ANS(I,J).EQ.9.) GOTO 1
C calculation of solar radiation
      TTAC=TATM(I,J)                    ! air temperature (C)
      TTA=TTAC+273.15                   ! air temperature (K)
      PPATM=PATM(I,J)*100.              ! atm.pressure (Pask)
      RATM=PPATM/287./TTA               ! air density
      TWAC=T1(I,J,1)                    ! water temperature (C)
      TWA=TWAC+273.15                   ! water temperature (K)
      HHUM=HUMID(I,J)                   ! humidity (parts of unit)
      WW=WIND(I,J)                      ! geostr. wind  (m/s)
      CCLO=CLOUD(I,J)                   ! cloudiness
      A=S1(I,J,1)
      DO K=1,NGR
      ANP(K)=AN1(I,J,K+1)
      IF(ANP(K).NE.0.) THEN
      HICP(K)=WICE1(I,J,K)/ANP(K)
      HSNP(K)=HSNOW(I,J,K)
      ELSE
      HICP(K)=0.
      HSNP(K)=0.
      ENDIF
      ENDDO
                                         !DO K=1,NGR1
      SPAR=A                             !SPAR(K)=A
      TPAR=TWA                           !TPAR(K)=TWA
                                         !ENDDO
      ANSUM=ANS(I,J)
      TFR=-54.*SPAR(1)                  ! freezing temperature (C)
      FW=1028.*4.1868E3*SKT(I,J)*(T1(I,J,1)-TFR)/DZ(1)
      IF(FW.LE.0.) FW=0.
      TFR=TFR+273.15                    ! freezing temperature (K)
      A1=FI(I,J)/57.3                   ! constants for solar radiation
      A2=SIN(A1)
      B2=COS(A1)
      B=23.44*COS((172-NDAY)/57.3)/57.3
      A3=SIN(B)
      B3=COS(B)
      CZ=0.
C      DO K=III-7,III       ! THAT IS FOR 1-HOUR TIME STEP DT
      B1=A2*A3+B2*B3*COS((12-K)*0.2617994) ! THAT IS AS WELL
      B1=A2*A3+B2*B3*COS((12-(III*8-4))*0.2617994)
      IF(B1.LT.0.) B1=0.
      CZ=CZ+B1
C      ENDDO
C      CZ=CZ/8.              ! THAT IS FOR 1-HOUR TIME STEP DT
      A1=1353.*CZ*CZ*(1.-0.6*CCLO*CCLO*CCLO)
      B1=(CZ+2.7)*1.E-5
      B2=1.085*CZ+0.1
      E=HHUM*610.78*10**(8.61503*(TTA-273.15)/TTA) ! vapor pressure
      SW=A1/(B1*E+B2)                  ! solar radiation (f-lae 1.4,1.5)
      B3=RATM*1.7068*WW                ! constant for SH
      A3=0.6650735*RATM/PPATM*WW       ! constant for EL
      WL=5.4999E-8*TTA**4*(1.-0.261*EXP
     *(-7.77E-4*((273.15-TTA)**2)))*
     *(1.+0.275*CCLO)                  ! incoming longwave rad.F (f-la 1.3)
C+++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++
C     fluxes and possible formation of ice (no ice is initial)
C+++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++
      HFIRST=0.
      ANN1=AN1(I,J,1)
      IF(ANN1.EQ.0.) GOTO 11
      SH1=B3*(TTA-TWA)                            ! SH (f-la 1.2)
      EL1=A3*2.5E6*(HHUM*10**(7.63*TTAC/(241.9+TTAC))-
     *10**(7.63*TWAC/(241.9+TWAC)))               ! LE (f-la 1.1)
      AAA=-0.97*5.67E-8*(TWA**4)
      QN=SH1+EL1+0.97*WL+0.9*SW-0.97*5.67E-8*(TWA**4)
C
      IF(QN.GT.0.) THEN
      TPAR(1)=TPAR(1)+DTZ*ANN1*QN                 ! ANN1 - for leads
      ELSE
      TPAR(1)=TPAR(1)+DTZ*QN
      ENDIF
      IF(TPAR(1).GE.TFR) THEN
      IF(TPAR(1).GT.TTA.AND.QN.GT.0.) TPAR(1)=TTA ! no ice formation
      ELSE
      TPAR(1)=TFR                                 !  ice formation
      HFIRST=0.01
      ENDIF
C
   11 CONTINUE
C@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@
C                     fluxes with ice
C@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@
      DO 20 K=1,NGR
      K1=K+1
      IF(ANP(K).LT.0.001) GOTO 20
C
      DHSN=0.
      DHIC1=0.
      TTS=0.5*(TTA+TWA)
      EL=A3*2.834E6*(HHUM*10**(9.5*TTAC/(265.5+TTAC))-
     *10**(9.5*(TTS-273.15)/(TTS-7.65)))      ! EL (f-la 1.1)
      HHIC1=1./HICP(K)
      IF(HSNP(K).NE.0.) GOTO 30
C****************** ice without snow    *******************
      B=2.04*HHIC1
      ERR=100.
      KU=O
      DO WHILE(ERR.GE.0.25)
      SH=B3*(TTA-TTS)                         ! SH (f-la 1.2)
      A=5.4999E-8*TTS**3                      ! 5.4999E-8=(0.97*5.67E-8)
      A1=SH+EL+0.466*SW+0.97*WL
      DTTS=(A1-A*TTS+B*(TFR-TTS))/(4.*A+B)
      A=TTS+DTTS                              ! new surface temperature
      ERR1=ERR
      ERR=ABS(DTTS)
      IF(ERR.GE.ERR1) GOTO 40
      TTS=A
      ENDDO
      GOTO 41
   40 TTS=TTA
   41 CONTINUE
C****  processes at the surface "ice-air"
      IF(TTS.LE.273.15) GOTO 25
C    melting at the ice surface
      TTS=273.15
      DHIC1=-DT/302.E6*(A1-5.4999E-8*TTS**4+B*(TFR-TTS))
      HICP(K)=HICP(K)+DHIC1
      IF(HICP(K).LT.0.01) HICP(K)=0.
      GOTO 26
C    snow fall parameterization
   25 DHSN=DT*SFAL(LLL)
      HSNP(K)=DHSN
C****  processes at the surface "ice-water"
   26 DHIC=-DT/302.E6*(FW-B*(TFR-TTS))
      HICP(K)=HICP(K)+DHIC
      DHIC=DHIC+DHIC1
      IF(HICP(K).LT.0.01) THEN
      HICP(K)=0.
      HSNP(K)=0.
      ENDIF
      GOTO 21
C******************  ice with snow ************************
   30 CONTINUE
      B1=0.31*HICP(K)
      B2=2.04*HSNP(K)
      B4=B1+B2
      B=0.6324/B4                           ! 0.6324=2.04*0.31
      ERR=100.
      DO WHILE(ERR.GE.0.25)
      SH=B3*(TTA-TTS)                       ! SH (f-la 1.2)
      A=5.6133E-8*TTS**3                    ! 5.6133E-8=(0.99*5.67E-8)
      A1=SH+EL+(1.-ALSN(LLL))*SW+0.99*WL
      DTTS=(A1-A*TTS+B*(TFR-TTS))/(4.*A+B)
      A=TTS+DTTS                            ! new snow Tsur
      ERR1=ERR
      ERR=ABS(DTTS)
      IF(ERR.GE.ERR1) GOTO 45
      TTS=A
      ENDDO
      GOTO 46
   45 TTS=TTA
   46 CONTINUE
C**** processes at the surface "snow-air"
      IF(TTS.LE.273.15) THEN
      DHSN=SFAL(LLL)*DT                             ! snow fall
      TTI=(B1*TTS+B2*TFR)/B4                        ! new ice Tsfc
      ELSE
      TTS=273.15                                    !snow melting
      TTI=(B1*273.15+B2*TFR)/B4                     ! new ice Tsfc
      DHSN=-DT/110.E6*(A1-5.6133E-8*TTS**4+
     *0.31/HSNP(K)*(TTI-TTS))                       ! f-la (2.14)
      ENDIF
C**** processes at the surface "ice-water"
      DHIC=DT/302.E6*(2.04*(TFR-TTI)/HICP(K)-FW)    ! f-la (2.15) !!!!!!!!!
      HICP(K)=HICP(K)+DHIC
      IF(HICP(K).GE.0.01) THEN
      A=0.1*HICP(K)
      HSNP(K)=HSNP(K)+DHSN
      IF(HSNP(K).GT.A) THEN
      HSNP(K)=A
      ELSE
      IF(HSNP(K).LT.0.) HSNP(K)=0.
      ENDIF
      ELSE
      HICP(K)=0.
      HSNP(K)=0.
      ENDIF
C********* new water temperature and salinity values *********
   21 A=0.88*DHIC
      B=DZZ-A
      IF(DHIC.LT.0.) THEN
         IF(DHSN.LT.0.) THEN                          ! melting of
         A1=0.3*DHSN                                  ! ice + snow
         B=B-A1
         B1=273.15*A1
         ELSE
         B1=0.
         A1=0.
         ENDIF
         SPAR(K1)=(SPAR(K1)*(DZZ+A+A1)-SICST(K)*A)/DZZ   ! when melting
         TFR=-54.*SPAR(K1)+273.15
         TPAR(K1)=(TFR*DZZ-B1)/(DZZ-A1)                  ! when melting
      ELSE
         SPAR(K1)=(SPAR(K1)*DZZ-SICST(K)*A)/B         ! when freezing
         TFR=-54.*SPAR(K1)+273.15
         TPAR(K1)=TFR
      ENDIF
      HSNOW(I,J,K)=HSNP(K)
      HICE(I,J,K)=HICP(K)
      IF(HICP(K).EQ.0.) THEN
      ANN1=ANN1+ANP(K)
      AN1(I,J,K1)=0.
      ANP(K)=0.
      ENDIF
   20 CONTINUE
C
      IF(ANN1.GT.0..AND.ANN1.LT.1..AND.QN.GT.0.) THEN
C************ decreasing of ice area due to melting in leads *******
         A=ANN1*QN*DT
         B=0.
         A3=0.
         B3=0.
         DO 55 K=1,NGR
         IF(HICP(K).NE.0.) THEN
         K1=K+1
         DANP(K)=ANP(K)*A/(302.E6*HICP(K)+110.E6*HSNP(K))   ! f-la 3.1
         ANP(K)=ANP(K)-DANP(K)                              ! f-la 3.2
         B=B+DANP(K)
         A3=A3+ANP(K)*TPAR(K1)
         B3=B3+ANP(K)*SPAR(K1)
         ELSE
         DANP(K)=0.
         ENDIF
   55    CONTINUE
         ANN2=ANN1+B                                        ! f-la 3.3
         A2=0.
         B2=0.
         DO 60 K=1,NGR
         K1=K+1
         TFR=-54.*SICST(K)+273.15                           ! f-la 3.5
         A=0.88*HICP(K)
         A1=DANP(K)/(A+DZZ)
         A2=A2+(TPAR(K1)*DZZ+TFR*A)*A1                      ! f-la 3.4
         B2=B2+(SPAR(K1)*DZZ+SICST(K)*A)*A1                ! f-la 3.4
   60    CONTINUE
         TPAR(1)=(TPAR(1)*ANN1+A2)/ANN2                     ! f-la 3.6
         SPAR(1)=(SPAR(1)*ANN1+B2)/ANN2                     ! f-la 3.6
         AN1(I,J,1)=ANN2
         T1(I,J,1)=ANN2*TPAR(1)+A3-273.15                   ! f-la 3.7
         S1(I,J,1)=ANN2*SPAR(1)+B3                          ! f-la 3.7
      ELSE
C****  new water temperature and salinity without lateral melting ***
         A1=ANN1*TPAR(1)
         B1=ANN1*SPAR(1)
         DO K=1,NGR
         K1=K+1
         A1=A1+ANP(K)*TPAR(K1)
         B1=B1+ANP(K)*SPAR(K1)
         ENDDO
         T1(I,J,1)=A1-273.15
         S1(I,J,1)=B1
         IF(HFIRST.GT.0.) THEN
         A2=AN1(I,J,2)
         A1=ANN1+A2
         HICP(1)=(0.01*ANN1+HICP(1)*A2)/A1
         AN1(I,J,1)=0.
         ANP(1)=A1
         ENDIF
         ENDIF
C      AN1(I,J,1)=ANN1
      DO K=1,NGR
      HICE(I,J,K)=HICP(K)
      HSNOW(I,J,K)=HSNP(K)
      AN1(I,J,K+1)=ANP(K)
      IF(ANP(K).NE.0.) THEN
      WICE1(I,J,K)=HICE(I,J,K)*ANP(K)
      ELSE
      WICE1(I,J,K)=0.
      ENDIF
      ENDDO
C
    1 CONTINUE
C
      RETURN
      END


      subroutine REDIS(IS1,JS1,KT1,NGR,NGR1,WICE1,AN1,HICE,
     *WICES,ANS,HICES,HSNOW,HSNP,HST,HMAX)
      REAL WICE1(IS1,JS1,NGR),AN1(IS1,JS1,NGR1),HICE(IS1,JS1,NGR),

     *HST(NGR),HMAX(NGR),HSNP(NGR),ANPR(5),WICPR(5),HPR(5)

C*******

      NGR2=NGR-1
      IS=IS1-1
      JS=JS1-1
      DO 314 I=1,IS
      DO 314 J=1,JS
      IF(KT1(I,J).EQ.0) GOTO 314
C CALCULATION WITH H-GRADE CONTROL
      DO K=1,NGR
      HICE(I,J,K)=0.
      WICPR(K)=WICE1(I,J,K)
      ANPR(K)=AN1(I,J,K+1)
      HSNP(K)=HSNOW(I,J,K)
      IF(WICPR(K).LE.0..OR.ANPR(K).LE.0.) THEN
      WICPR(K)=0.
      ANPR(K)=0.
      HSNP(K)=0.
      ENDIF
      ENDDO
      DO K=1,NGR2
      K1=K+1
      IF(WICPR(K).NE.0.) THEN
      IF(ANPR(K).EQ.0.) ANPR(K)=WICPR(K)/HST(K)
      IF(WICPR(K)/ANPR(K).GT.HMAX(K)) THEN
      ANPR(K1)=ANPR(K1)+ANPR(K)
      WICPR(K1)=WICPR(K1)+WICPR(K)
      HSNP(K1)=HSNP(K1)+HSNP(K)
      WICPR(K)=0.
      ANPR(K)=0.
      HSNP(K)=0.
      ENDIF
      ENDIF
      ENDDO
C
      DO K=NGR,2,-1
      IF(WICPR(K).GT.0.) THEN
      K2=K-1
      IF(WICPR(K)/ANPR(K).LT.HMAX(K2)) THEN
      WICPR(K2)=WICPR(K2)+WICPR(K)
      ANPR(K2)=ANPR(K2)+ANPR(K)
      WICPR(K)=0.
      ANPR(K)=0.
      ENDIF
      ENDIF
      ENDDO
C
      DO K=1,NGR
      IF(ANPR(K).NE.0.) THEN
      HPR(K)=WICPR(K)/ANPR(K)
      ELSE
      HPR(K)=0.
      ENDIF
      ENDDO
C
  319 A1=0.
      DO K=1,NGR
      A1=A1+ANPR(K)
      ENDDO
      IF(A1.LE.1.) GOTO 315
      DD=A1-1.
      A3=WICPR(1)
      B3=1.
      DO K=2,NGR
      B3=B3-ANPR(K)
      ENDDO
      DO 317 K=1,NGR2
      K1=K+1
      A3=A3+WICPR(K1)
      B3=B3+ANPR(K1)
      IF(ANPR(K).EQ.0.) GOTO 317
      IF(ANPR(K1).EQ.0.) THEN
      IF(DD.LT.ANPR(K)) THEN
      ANPR(K)=ANPR(K)-DD
      GOTO 315
      ELSE
      WICPR(K1)=WICPR(K)
      ANPR(K1)=WICPR(K)/HST(K1)
      HPR(K1)=WICPR(K1)/ANPR(K1)
      WICPR(K)=0.
      ANPR(K)=0.
      HPR(K)=0.
      ENDIF
      GOTO 319
      ENDIF
      A=HPR(K)-HPR(K1)
      B2=(A3-HPR(K1)*B3)/A
      A2=(HPR(K)*B3-A3)/A
          IF(B2.LT.0..OR.A2.LT.0.) THEN
          ANPR(K)=0.
          ELSE
          ANPR(K)=B2
          ANPR(K1)=A2
          GOTO 315
          ENDIF
  317 CONTINUE
      ANPR(NGR)=1.
      WICPR(NGR)=A3
  315 A1=0.
      B1=0.
        DO K=1,NGR
        IF(ANPR(K).EQ.0.) WICPR(K)=0.
        A1=A1+ANPR(K)
        B1=B1+WICPR(K)
        AN1(I,J,K+1)=ANPR(K)
        WICE1(I,J,K)=WICPR(K)
        HSNOW(I,J,K)=HSNP(K)
        IF(ANPR(K).NE.0.) HICE(I,J,K)=WICPR(K)/ANPR(K)
        ENDDO
C
      DO K=NGR,2,-1
      K2=K-1
      IF(HICE(I,J,K).LT.HMAX(K2)) THEN
      K1=K+1
      A=AN1(I,J,K)+AN1(I,J,K1)
      AN1(I,J,K)=A
      B=WICE1(I,J,K2)+WICE1(I,J,K)
      WICE1(I,J,K2)=B
      IF(A.NE.0.) HICE(I,J,K2)=B/A
      AN1(I,J,K1)=0.
      WICE1(I,J,K)=0.
      HICE(I,J,K)=0.
      ENDIF
      ENDDO
C
      ANS(I,J)=A1
      WICES(I,J)=B1
      IF(A1.GE.0.005) THEN
      HICES(I,J)=B1/A1
      ELSE
      ANS(I,J)=0.
      HICES(I,J)=0.
      WICES(I,J)=0.
      ENDIF
      AN1(I,J,1)=1.-A1
  314 CONTINUE
C=========================================================================
      RETURN
      END

      subroutine STRESS(IS,JS,IS1,JS1,IJS1,IS3,JS3,NGR,NGR1,DX,DT1,KT1,
     *ANS,AN1,HST,U,V,CD2,EXX,EYY,EXY,EPR,SXX,SYY,SXY)
c

     *EXX(IS1,JS1),EYY(IS1,JS1),EXY(IS1,JS1),
     *EPR(IS3,JS3),U(IS1,JS1),V(IS1,JS1),CD2(IS1,JS1),
     *SXX(IS1,JS1),SYY(IS1,JS1),SXY(IS1,JS1)
      integer KT1(IS1,JS1)

      ENDDO

      DO I=1,IS
      I1=I+1

      J1=J+1

      A=EXX(I,J)+EYY(I,J)

      ASA1=FKU(I,J)*0.5*DT
      ASA=1.+ASA1*ASA1
C      RI2J=RO(I2,J,1)
C      RIJ=RO(I,J,1)

C      RIJ2=RO(I,J2,1)

      DZZ=DZ(1)
      ADZ1=W(I2,J2,1)+W(I2,J,1)+W(I,J2,1)+W(I,J,1)
C******
      DO 201 K=1,KI
      K2=K-1
      IF(HHT.EQ.Z(K)) GOTO 202
      K1=K+1
      DZZ1=DZ(K1)
      UIJ=U1(I,J,K)
      VIJ=V1(I,J,K)
C**** integral of DRO/DX and DRO/DY calculation
C      RI2J=RO(I2,J,K)
C      RIJ2=RO(I,J2,K)
C      RIJ=RO(I,J,K)
C      RI2J2=RO(I2,J2,K)
C      A1=RI2J+RIJ-RI2J2-RIJ2
C      B1=RI2J2+RI2J-RIJ2-RIJ
C      CC=C8*DZZ
C      SUM=SUM+(A+A1)*CC
C      SUM1=SUM1+(B+B1)*CC
C*** Laplasian of  U' and  V'  calculation
      SLAPU=U1(I,J2,K)+U1(I,J1,K)+U1(I2,J,K)+U1(I1,J,K)-4.*UIJ
      SLAPV=V1(I1,J,K)+V1(I2,J,K)+V1(I,J1,K)+V1(I,J2,K)-4.*VIJ
C**********
      AUU=UIJ+ASA1*VIJ+DT*(-C1*SUM-DPX(I,J)+C3*SLAPU)
      AVV=VIJ-ASA1*UIJ+DT*(-C1*SUM1-DPY(I,J)+C3*SLAPV)
      U2(I,J,K)=(AUU+AVV*ASA1)/ASA
      V2(I,J,K)=(AVV-AUU*ASA1)/ASA
      A=A1
      B=B1
      DZZ=DZZ1
      GOTO 201
  202 U2(I,J,K)=0.
      V2(I,J,K)=0.
  201 CONTINUE
  200 CONTINUE
      write(*,*)' U & V'
      write(2,*)' U & V'

C*******   vertical viscosity *************

      DO 210 I=2,IS
      I2=I-1
      DO 210 J=2,JS
      KI=KK1(I,J)
      IF(KI.EQ.0) GOTO 210
      J2=J-1
      HHT=MAP(I,J)
      KI1=KI+1
      KI2=KI-1

C****** calculation of viscosity coefficients NU=L*L*dU/dz

      YYY=0.25*(YM2(I2,J2)+YM2(I,J2)+YM2(I2,J)+YM2(I,J))
      UIJ=U2(I,J,1)
      VIJ=V2(I,J,1)
      A=SQRT(UIJ*UIJ+VIJ*VIJ)
      DO 211 K=1,KI
      K1=K+1
      UIJ=U2(I,J,K1)
      VIJ=V2(I,J,K1)
      B=SQRT(UIJ*UIJ+VIJ*VIJ)
      FF=HHT-Z(K)+2.
      FF1=Z(K)-YYY-2.
      SL=0.4/HHT*FF*FF1*(1.-1.2*FF*FF1/HHT/HHT)/(4.*FF1/5000.+1.)
      IF(K.NE.KI) THEN
      RR(K)=SL*SL*ABS(B-A)/DZ(K1)
      ELSE
      RR(K)=SL*SL*ABS(B-A)/(HHT-Z(KI)+50.)
      ENDIF
      A=B
  211 CONTINUE
      SKZ(I,J)=RR(1)
C       d i r e c t    s t e p
      IF(KI.NE.1) THEN
      DZZ=DZ1(1)
      DZZZ=DZ(2)
      ELSE
      DZZ=HHT
      DZZZ=HHT-Z(1)+50.
      ENDIF
      B=DT/DZZ/DZZZ*RR(1)
      A=1.+B
      UCA(1)=B/A
      A1=0.25*(ANS(I,J)+ANS(I,J2)+ANS(I2,J)+ANS(I2,J2))
      UNU(1)=(U2(I,J,1)+DT/DZZ*((1.-A1)*TX(I,J)+A1*TXIC(I,J)))/A
      VCA(1)=UCA(1)
      VNU(1)=(V2(I,J,1)+DT/DZZ*((1.-A1)*TY(I,J)+A1*TYIC(I,J)))/A
      IF(KI.EQ.1) GOTO 212
      IF(KI.EQ.2) GOTO 214
      DO 215 K=2,KI2
      K2=K-1
      AA=DT/DZ1(K)
      A=-AA/DZ(K)*RR(K2)
      B=-AA/DZ(K+1)*RR(K)
      A1=-1.+A+B
      AA=A1-UCA(K2)*A
      UCA(K)=B/AA
      UNU(K)=(A*UNU(K2)-U2(I,J,K))/AA
      AA=A1-VCA(K2)*A
      VCA(K)=B/AA
  215 VNU(K)=(A*VNU(K2)-V2(I,J,K))/AA
C    i n v e r s e    s t e p
  214 DZZ=HHT-0.5*(Z(KI)+Z(KI2))
      DZZZ=HHT-Z(KI)+50.
      B=DT/DZZ/DZ(KI)*RR(KI2)
      A=1.+B+DT/DZZ/DZZZ*RR(KI)
      BB=B/A
      U2(I,J,KI)=(U2(I,J,KI)/A+BB*UNU(KI2))/(1.-BB*UCA(KI2))
      V2(I,J,KI)=(V2(I,J,KI)/A+BB*VNU(KI2))/(1.-BB*VCA(KI2))
      DO 213 K=1,KI2
      KK=KI-K
      K1=KK+1
      U2(I,J,KK)=UCA(KK)*U2(I,J,K1)+UNU(KK)
  213 V2(I,J,KK)=VCA(KK)*V2(I,J,K1)+VNU(KK)
      GOTO 210
  212 U2(I,J,KI)=UCA(KI)*U2(I,J,KI1)+UNU(KI)
      V2(I,J,KI)=VCA(KI)*V2(I,J,KI1)+VNU(KI)
  210 CONTINUE
      write(2,*)' vertical viscosity'
      write(*,*)' vertical viscosity'
C========================================================================
C    calculation of the vertically mean currents
C========================================================================
      CALL shal(IS,JS,KS,IS1,JS1,IS2,JS2,LI,LJ,DT1,DX,C1,C3,C9,C10,MM3,
     *HT,HU,HV,KT1,KUSH,KVSH,Y2,UP1,UP2,VP1,VP2,YM2,

      write(2,*)' shallow'

      DO 280 I=2,IS

      J2=J-1

      IF(KI.NE.1) THEN
      DZZ=HHT-0.5*(Z(KI)+Z(KI-1))

      DZZ=HHT
      ENDIF
      ELSE
      DZZ=DZ1(K)
      ENDIF
      SUM=U2(I,J,K)*DZZ+SUM
      SUM1=V2(I,J,K)*DZZ+SUM1

      SUM=(-SUM+0.5*(UP2(I,J)+UP2(I2,J)))/HHT
      SUM1=(-SUM1+0.5*(VP2(I,J)+VP2(I,J2)))/HHT
      DO K=1,KI
      U2(I,J,K)=U2(I,J,K)+SUM
      V2(I,J,K)=V2(I,J,K)+SUM1

  280 CONTINUE

  333 CONTINUE

        IF(LLK.EQ.12)THEN
